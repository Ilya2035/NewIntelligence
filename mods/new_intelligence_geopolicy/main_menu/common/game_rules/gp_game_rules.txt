# New Intelligence: Geopolicy - game rules (prefix gp_)
# Format: game/main_menu/common/game_rules/_game_rules.info and 00_game_rules.txt; same conventions as the Battle mod:
# * every setting carries flag = general_rule (otherwise invisible in the game-rules UI) and a flag named like itself,
#   so script can test has_game_rule = <setting key> either way.
# * Loc keys: rule_<rule key>, setting_<setting key>, setting_<setting key>_desc.
# * NO "defines = { }" block inside a setting: documented in _game_rules.info, but the 1.4.7 parser rejects it
#   ("Unexpected token: defines", Battle first launch 2026-11-04). The define overrides of
#   loading_screen/common/defines/gp_defines.txt are therefore always active.

# Geopolicy AI: standard / strong / off. Read by gp_is_active / gp_is_strong_mode (scripted_triggers/gp_triggers.txt).
# Off keeps the defines and the formable countries but applies no roles, stages, threat reactions or war scoring.
gp_rule_mode = {
	default = gp_mode_standard

	gp_mode_standard = {
		flag = gp_mode_standard
		flag = general_rule
	}
	gp_mode_strong = {
		flag = gp_mode_strong
		flag = general_rule
	}
	gp_mode_off = {
		flag = gp_mode_off
		flag = general_rule
	}
}

# v0.12.1 The foreign invader (in_game/common/scripted_effects/gp_invader_effects.txt, scripted_triggers/gp_invader_triggers.txt).
# Author, 2026-11-10: "an additional temporary united alliance against Morocco". A power from another region, another macro-region
# and another faith group that fights the leaders of a region (or holds three of its locations) is the regional threat of every
# country in it: the neighbours close ranks (common front posture, alliance reasons, opinion, the war scores), the rivals of the
# region lay their feud aside while it lasts. After two lost wars within thirty years the invader loses interest in that region for
# 25 years. Where it works at all: the rule gp_rule_invader_scope below (Iberia by default).
#   pact  (default) the AI countries of such a front are also ALLIED BY THE SCRIPT for the length of the crisis and their rivalries
#                   are removed (create_alliance / remove_rival; a crutch - it lapses when the invader no longer threatens them);
#   front           the evaluation alone: alliances only by the AI's own reasons (the weights of gp_alliance.txt);
#   off             no invader: the regional threat is found by strength and neighbourhood only.
# Code tests: gp_invader_off (everything), gp_invader_front / gp_invader_off (the pact, gp_inv_pact_enabled).
gp_rule_invader = {
	default = gp_invader_pact

	gp_invader_pact = {
		flag = gp_invader_pact
		flag = general_rule
	}
	gp_invader_front = {
		flag = gp_invader_front
		flag = general_rule
	}
	gp_invader_off = {
		flag = gp_invader_off
		flag = general_rule
	}
}

# v0.12.1 Where the foreign invader works (independent review of the first build, 2026-11-10). The author asked for Morocco and Iberia
# only; the first build was worldwide and, by the reviewers' reading of the 1337 setup (never seen in the game), would have flagged
# Byzantium as the invader of 19 Anatolian beyliks (the Ottomans among them) and Denmark as the invader of pagan Lithuania on the
# first pass, and would have allied the beyliks with one another by script.
# Staged rollout: Iberia first, the world after the first observation.
#   iberia (default) only countries with their capital in the Iberian region have invaders (Morocco against Castile, Aragon,
#                    Portugal, Navarre; the war arm, the foothold arm and the record of lost wars all look at that region alone);
#   world            every region (Ottomans against the Balkans, Mamluks against Cilicia, ...) - NOT YET OBSERVED.
# Code tests: gp_inv_in_scope (scripted_triggers/gp_invader_triggers.txt).
gp_rule_invader_scope = {
	default = gp_invader_iberia

	gp_invader_iberia = {
		flag = gp_invader_iberia
		flag = general_rule
	}
	gp_invader_world = {
		flag = gp_invader_world
		flag = general_rule
	}
}
