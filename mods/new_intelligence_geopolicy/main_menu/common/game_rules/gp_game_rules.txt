# New Intelligence: Geopolicy - game rules (prefix gp_)
# Format: game/main_menu/common/game_rules/_game_rules.info and 00_game_rules.txt; same conventions as the Battle mod:
# * every setting carries flag = general_rule (otherwise invisible in the game-rules UI) and a flag named like itself,
#   so script can test has_game_rule = <setting key> either way.
# * Loc keys: rule_<rule key>, setting_<setting key>, setting_<setting key>_desc.
# * NO "defines = { }" block inside a setting: documented in _game_rules.info, but the 1.4.7 parser rejects it
#   ("Unexpected token: defines", Battle first launch 2026-11-04). The define overrides of
#   loading_screen/common/defines/gp_defines.txt are therefore always active.

# Geopolicy AI: standard / strong / off. Read by gp_is_active / gp_is_strong_mode (scripted_triggers/gp_triggers.txt).
# Off keeps the defines and the formable countries but applies no roles, stages, threat reactions or war scoring.
gp_rule_mode = {
	default = gp_mode_standard

	gp_mode_standard = {
		flag = gp_mode_standard
		flag = general_rule
	}
	gp_mode_strong = {
		flag = gp_mode_strong
		flag = general_rule
	}
	gp_mode_off = {
		flag = gp_mode_off
		flag = general_rule
	}
}
