###################################
# New Intelligence: Geopolicy - country modifiers (prefix gp_)
#
# Applied to AI countries only (gp_update_role / gp_apply_temperament, in_game/common/scripted_effects/gp_effects.txt)
# with months = 14 and mode = replace, refreshed yearly; they self-expire if the mod stops refreshing them.
# Every modifier type here is marked ai = yes in modifier_type_definitions (strategic valuation only; players never
# feel them) except ai_require_cb_for_war, which is AI-only by name (vanilla static_modifiers/difficulty.txt).
# Value ranges follow vanilla: ai personalities +-0.3 aggressiveness / +-0.25 carefulness, difficulty high_aggression +1.0.
# The keys are visible on the AI country's "Behavioral Modifiers" tooltip, so the names read as the role
# (STATIC_MODIFIER_NAME_<key> / STATIC_MODIFIER_DESC_<key>, gp_geopolicy_l_*.yml).
###################################

# ---- roles (exactly one per AI country) --------------------------------------------------------

# Small or fragmented: takes its own area and region, wants land rather than vassals, strikes when the moment comes.
gp_role_consolidator = {
	game_data = { category = country }
	aggressiveness_modifier = 0.15
	carefulness_modifier = -0.05
	ai_months_between_wars = -6
	win_war_chance_threshold = -0.05
	coalition_strength_tolerance = 0.05
	subjugation_preference_modifier = -0.1
	unintegrated_land_expansion_penalty_modifier = -0.25
}
