# The African interior (inland, unowned, south of the Maghreb and Egypt): settlers come slowly. Removed once the land is
# someone's overseas possession.

	game_data = { category = location }
	local_migration_speed_modifier = -0.5
}

	land_morale_modifier = 0.05
}

# A power from outside Asia present in more than two zones of Asia: every new conquest there angers its neighbours more
# and takes longer to integrate.

	game_data = { category = country }
	antagonism_taking_land_giving_modifier = 0.5
	global_integration_speed_modifier = -0.25
}
