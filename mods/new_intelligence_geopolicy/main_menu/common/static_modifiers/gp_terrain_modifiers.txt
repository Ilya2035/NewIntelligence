# A fortress in the mountains: twice as hard to take, garrison refills faster.
gp_mountain_stronghold = {
	game_data = { category = location }
	local_defensive = 1.0
	local_garrison_growth = 0.5
}

# Hills: a little harder.
gp_hill_stronghold = {
	game_data = { category = location }
	local_defensive = 0.3
	local_garrison_growth = 0.15
}

# Open steppe: nothing to hold on to.
gp_open_steppe = {
	game_data = { category = location }
	local_defensive = -0.25
	local_garrison_growth = -0.25
}
