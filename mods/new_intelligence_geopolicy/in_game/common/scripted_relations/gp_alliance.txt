# New Intelligence: Geopolicy - the vanilla alliance (in_game/common/scripted_relations/alliance.txt, 1.4.7) with the
# reasons of the region added to the AI's own weighing (game rule gp_rule_alliance):
#   gp_alliance_reasons (default, v0.12) - no opinion threshold; the country asked weighs, next to the vanilla reasons
#     (opinion, trust, needs, common rivals, desire for our land), the reasons of Geopolicy (wants_to_give): rivals for
#     the same region (-60), the partner is our own regional threat (-50), the partner is ruined (broken -60, in crisis
#     -40, freshly defeated -20), a partner from another region with no common enemy (-30), before 1450 a partner from
#     outside the western game or into it (-40), a common regional threat (+15). An ally keeps an alliance by the same
#     reasons, milder (wants_to_keep). Author, 2026-11-09: "the AI must make the right decision itself, not be held by
#     frames" - the opinion-100 gate of v0.9 was such a frame (docs/GEOPOLICY_V012_PLAN.md, table 3a);
#   gp_alliance_friends - the same reasons and, on top, the v0.9 gate: opinion 100 both ways to offer;
#   gp_alliance_any - the vanilla alliance.
# v0.12.1 (observation 1337-1466): a FOREIGN INVADER (gp_ally_invader_front: both countries name the same foreign power as
# their regional threat) is a reason of its own, +60 to make an alliance and +40 to keep it (GP_ALLY_INVADER); it replaces the
# flat common-threat reason, cancels the rivalry of one region and the ruin of a partner beaten by the same invader (the
# alliances of Castile, Aragon and Portugal against Morocco - the evaluation route; the script pact is a rule of its own,
# gp_rule_invader, gp_invader_effects.txt). The common regional threat scales with its size (+15 at 1.5x, +25 at 2x, +35 at 3x),
# and the old friendships of the history are renewed (+20 to make, +10 to keep: Poland - Hungary, Poland - Lithuania).
# Existing alliances are not broken by the rule itself (the vanilla will_expire_trigger is unchanged). Re-check against
# vanilla after every game patch.
REPLACE:alliance = {
	type = diplomacy
	relation_type = mutual
    uses_diplo_capacity = mutual
    block_when_at_war = yes
    break_on_war = yes
    break_on_becoming_subject = yes
    annulled_by_peace_treaty = yes
    annullment_favours_required = 20
    disallow_war = yes
    military_access = yes
	can_share_maps = yes
    fleet_basing_rights = yes
    lifts_fog_of_war = yes
    called_in_defensively = mutual
    called_in_offensively = mutual
    mutual_color = define:NMapColors|DIPLOMACY_ALLIANCE_COLOR
    category = {
        offer = CATEGORY_FRIENDLY_ACTIONS
        cancellation = CATEGORY_HOSTILE_ACTIONS
        request = CATEGORY_FRIENDLY_ACTIONS
        break = CATEGORY_HOSTILE_ACTIONS
    }

    relation_type_for_ai = friendly
    show_break_alert = yes
	
	
	diplomatic_capacity_cost = alliance_upkeep_cost

    favors_to_first = {
        add = {
            desc = "ALLIANCE_FAVORS_BASE"
            value = 1
        }
        add = {
            desc = "ALLIANCE_FAVORS_RELATIVE_STRENGTH"
            value = "scope:first.relative_military_strength(scope:second)"
        }
        divide = 12
    }

    favors_to_second = {
        add = {
            desc = "ALLIANCE_FAVORS_BASE"
            value = 1
        }
        add = {
            desc = "ALLIANCE_FAVORS_RELATIVE_STRENGTH"
            value = "scope:second.relative_military_strength(scope:first)"
        }
        divide = 12
    }

    institution_spread_to_first = monthly_institution_spread_mild
    institution_spread_to_second = monthly_institution_spread_mild

	offer_enabled = {
        alliance_can_exist = {
            first = scope:actor
            second = scope:recipient
        }
        OR = {
            NOT = { has_game_rule = gp_alliance_friends }
            custom_tooltip = {
                text = gp_alliance_needs_friendship_tt
                scope:actor = { opinion = { target = scope:recipient value >= 100 } }
                scope:recipient = { opinion = { target = scope:actor value >= 100 } }
            }
        }
	}

	request_visible = { always = no }
	break_visible = { always = no }

    cancel_enabled = {
        scope:actor = {
            not = { is_fighting_war_together_with = scope:recipient }
        }
    }

    will_expire_trigger = {
        not = {
            alliance_can_exist = {
                first = scope:first
                second = scope:second
            }
        }
    }

	offer_declined_effect = {
		scope:actor = {
			add_opinion = { target = scope:recipient modifier = opinion_decline_alliance }
		}
	}

    offer_effect = {
        #no need for guarantees with an alliance
        if = {
            limit = {
                scope:actor != scope:recipient
                scope:actor = {
                    giving_scripted_relation = {
                        type = relation_type:guarantee
                        target = scope:recipient
                    }
                }
            }
            scope:actor = {
                remove_relation = { 
                    first = scope:actor
                    second = scope:recipient
                    type = relation_type:guarantee
                }
            }
        }
        if = {
            limit = {
                scope:actor != scope:recipient
                scope:actor = {
                    receiving_scripted_relation = {
                        type = relation_type:guarantee
                        target = scope:recipient
                    }
                }
            }
            scope:actor = {
                remove_relation = { 
                    first = scope:recipient
                    second = scope:actor
                    type = relation_type:guarantee
                }
            }
        }
    }

    cancel_effect = {
        scope:actor = {
            custom_tooltip = cancel_alliance_custom_tt
            add_truce_with = {
                target = scope:recipient
                mutual = yes
            }
        }
        scope:recipient = {
            add_opinion = {
                target = scope:actor
                modifier = opinion_broke_alliance
            }
        }
        scope:recipient = {
            add_trust_equilibrium = {
                target = scope:actor
                modifier = trust_broke_alliance
            }
        }
    }

    should_ai_offer_trigger = {
    	NOT = {
	        scope:first = {
	            has_opinion = {
	                modifier = opinion_broke_alliance
	                target = scope:second
	            }
	        }
        }
    }

    expire_effect = {
        #downgrade to guarantee
        if = {
            limit = {
                scope:first = {
                    modifier:allowed_guarantee = yes
                    country_rank_level > scope:second.country_rank_level
                }
            }
            scope:first = {
                create_relation = {
                    first = scope:first
                    second = scope:second
                    type = relation_type:guarantee
                }
            }
        }
        else_if = {
            limit = {
                scope:second = {
                    modifier:allowed_guarantee = yes
                    country_rank_level > scope:first.country_rank_level
                }
            }
            scope:first = {
                create_relation = {
                    first = scope:second
                    second = scope:first
                    type = relation_type:guarantee
                }
            }
        }
    }

    wants_to_give_diplo_chance = {
        actor_is_rival = -100
        actor_overlord_is_rival = -100
        base = -40
        trust_in_actor = 1
        positive_opinion = 0.3
        negative_opinion = -1
        border_distance = -0.05
        few_relations = 5
        same_court_language = 15
        royal_ties = 3
        strategic_interest = 1
        common_threat = 0.4
        relative_strength = 25
        threatening_rivals = 20
    }
    wants_to_give = {
        scope:actor = {
            add = {
                desc = "COALITION_GRADE_ANTAGONISM_TOWARDS_THEM"
                value = {
                    every_country_with_coalition_grade_antagonism_against_us = {
                        limit = {
                            not = { is_subject_or_below_of = scope:actor }
                        }
                        subtract = {
                            desc = "[country_strength|e]"
                            value = country_strength
                            #countries further away mean less
                            divide = {
                                value = 1
                                add = {
                                    value = "border_distance_to(scope:actor)"
                                    divide = 600
                                }
                                min = 1
                            }
                        }
                    }
                    if = {
                        limit = {
                            NOT = {
                                country_strength <= 0
                            }
                        }
                        divide = country_strength
                    }
                    multiply = 10
                }
            }
        }
        if = {
            limit = { exists = scope:recipient }
            if = {
                #Actor needs to be close and relativly big
                limit = {
                    scope:recipient = {
                        border_distance_to = {
                            country = scope:actor
                            value < 150
                        }
                        relative_strength = {
                            target = scope:actor
                            value < 0.60
                        }
                    }
                }
                add = {
                    desc = "NEEDS_ALLIES_AGAINST_STRONG_RIVALS_AND_NEIGHBOURS"
                    # Check if the country needs allies at all
                    value = scope:recipient.country_needs_allies_value
                    max = 25
                    min = 0
                }
            }
            if = {
                limit = {
                    scope:actor = {
                        any_current_war = {
                            NOT = {
                                is_on_same_side = {
                                    country = scope:actor
                                    target = scope:recipient
                                }
                            }
                        }
                    }
                }
                add = {
                    desc = "ACTOR_IS_IN_DIFFERENT_WARS"
                    value = -100
                }
            }

            if = {
                limit = {
                    "scope:recipient.opinion(scope:actor)" < 150
                    "scope:recipient.conquer_desire(scope:actor)" > 20
                    scope:recipient = {
                        is_neighbor_of = scope:actor
                    }
                }
                add = {
                    desc = "RECIPIENTS_DESIRES_ACTORS_LOCATIONS"
                    value = "scope:recipient.conquer_desire(scope:actor)"
                    multiply = -1
                }
                if = {
                    limit = {
                        scope:recipient = {
                            ai_personality = ai_personality:ai_expansionist
                        }
                    }
                    add = {
                        desc = "DESIRES_LOCATIONS_TT_DESC"
                        value = "scope:recipient.num_of_locations_with_high_conquer_desire(scope:actor)"
                        multiply = -5
                    }
                }
            }

            if = {
                limit = {
                    situation:rise_of_the_ottomans = { has_variable = strongest_beylik_variable }
                    situation:rise_of_the_ottomans = { has_variable = second_strongest_strongest_beylik_variable }
                    scope:actor ?= situation:rise_of_the_ottomans.var:strongest_beylik_variable
                    scope:recipient ?= situation:rise_of_the_ottomans.var:second_strongest_strongest_beylik_variable
                }
                add = {
                    value = -1000
                    desc = "CHALLENGER_BEYLIK"
                }
            }

            add = {
                desc = "RECIPIENT_WOULD_NEED_TO_PAY_FOR_DIPLOMACY"
                value = scope:recipient.diplomatic_capacity_without_maintenance
                subtract = scope:recipient.used_diplomatic_capacity
                subtract = "scope:recipient.diplomatic_capacity_of_new_relation(scope:actor|scope:scripted_relation_type)"
                if = {
                    limit = {
                        NOT = { scope:recipient.diplomatic_capacity_without_maintenance <= 0 }
                    }
                    divide = scope:recipient.diplomatic_capacity_without_maintenance
                }
                max = 0
                multiply = 100
            }

            add = {
                desc = "COMMON_RIVALS"
                value = "scope:recipient.num_of_common_rivals(scope:actor)"
                max = 3
                multiply = 5
            }

            # ---- v0.12 Geopolicy: the reasons of the region (the country asked = scope:recipient) ----
            if = {
                limit = { NOT = { has_game_rule = gp_alliance_any } }
                if = {
                    limit = {
                        scope:actor = { gp_is_regional_rival_of = { COUNTRY = scope:recipient } }
                        NOT = { gp_ally_overwhelming_threat = { WHO = scope:recipient PARTNER = scope:actor } }
                        NOT = { gp_ally_invader_front = { WHO = scope:recipient PARTNER = scope:actor } }	# v0.12.1 rivals of one region stand together against a foreign invader
                    }
                    add = {
                        desc = "GP_ALLY_REGION_RIVALS"
                        value = -60
                    }
                }
                if = {
                    limit = { scope:actor = { this ?= scope:recipient.var:gp_threat } }
                    add = {
                        desc = "GP_ALLY_OWN_THREAT"
                        value = -50
                    }
                }
                if = {
                    limit = { scope:actor = { has_variable = gp_sc_broken } }
                    add = {
                        desc = "GP_ALLY_RUINED"
                        value = -60
                    }
                }
                else_if = {
                    limit = {
                        scope:actor = { has_variable = gp_crisis var:gp_crisis >= 80 }
                        NOT = { gp_ally_invader_front = { WHO = scope:recipient PARTNER = scope:actor } }	# v0.12.1 beaten by the same invader: still a partner
                    }
                    add = {
                        desc = "GP_ALLY_RUINED"
                        value = -40
                    }
                }
                else_if = {
                    limit = {
                        scope:actor = { has_variable = gp_recent_defeat }
                        NOT = { gp_ally_invader_front = { WHO = scope:recipient PARTNER = scope:actor } }
                    }
                    add = {
                        desc = "GP_ALLY_RUINED"
                        value = -20
                    }
                }
                if = {
                    limit = {
                        gp_ally_far_without_cause = { WHO = scope:recipient PARTNER = scope:actor }
                        "scope:recipient.num_of_common_rivals(scope:actor)" < 1
                    }
                    add = {
                        desc = "GP_ALLY_FAR_NO_CAUSE"
                        value = -30
                    }
                }
                if = {
                    limit = {
                        current_date < 1450.1.1
                        gp_ally_crosses_western_game = { WHO = scope:recipient PARTNER = scope:actor }
                        NOT = { scope:recipient = { gp_shares_threat_with = { COUNTRY = scope:actor } } }
                    }
                    add = {
                        desc = "GP_ALLY_WEST_CLOSED"
                        value = -40
                    }
                }
                # v0.12.1 a foreign invader names both of us (gp_ally_invader_front, gp_invader_triggers.txt): the strongest reason
                # there is. Author, principle 5: "against a common strong enemy the neighbours unite - Portugal, Castile and
                # Aragon against Granada and Morocco". +60 offsets the rivalry (-60) and the ruin (-40 / -20) above, which are
                # waived for such a pair anyway.
                if = {
                    limit = { gp_ally_invader_front = { WHO = scope:recipient PARTNER = scope:actor } }
                    add = {
                        desc = "GP_ALLY_INVADER"
                        value = 60
                    }
                }
                # v0.12.1 the common threat scales with its size (was +15 for a 1.5x and a 3x threat alike): +15 at 1.5x, +25 at
                # 2x (share 0.67), +35 at 3x (share 0.75). Three plain branches of the same reason, as the reasons above.
                else_if = {
                    limit = {
                        scope:recipient = {
                            gp_shares_threat_with = { COUNTRY = scope:actor }
                            var:gp_threat_share >= 0.75
                        }
                    }
                    add = {
                        desc = "GP_ALLY_COMMON_THREAT"
                        value = 35
                    }
                }
                else_if = {
                    limit = {
                        scope:recipient = {
                            gp_shares_threat_with = { COUNTRY = scope:actor }
                            var:gp_threat_share >= 0.67
                        }
                    }
                    add = {
                        desc = "GP_ALLY_COMMON_THREAT"
                        value = 25
                    }
                }
                else_if = {
                    limit = { scope:recipient = { gp_shares_threat_with = { COUNTRY = scope:actor } } }
                    add = {
                        desc = "GP_ALLY_COMMON_THREAT"
                        value = 15
                    }
                }
                # v0.12.1 the friendships of the history (Poland - Hungary 1335, Poland - Lithuania 1325, England - Portugal) are
                # renewed when an alliance expires: the country asked remembers the actor as a friend (gp_is_old_friend_of).
                if = {
                    limit = { scope:actor = { gp_is_old_friend_of = { COUNTRY = scope:recipient } } }
                    add = {
                        desc = "GP_ALLY_OLD_FRIEND"
                        value = 20
                    }
                }
            }
        }
    }

    wants_to_keep_diplo_chance = {
        recipient_is_rival = -100
        recipient_overlord_is_rival = -100
        trust_in_recipient = 1
        border_distance = -0.05
        same_court_language = 15
        royal_ties = 3
        strategic_interest = 1
        common_threat = 0.4
        reverse_relative_strength = 25
        threatening_rivals = 20
    }
    wants_to_keep = {
        add = {
            desc = "DIPLOREASON_NEGATIVE_OPINION"
            value = "scope:actor.opinion(scope:recipient)"
            max = 0        
        }
        add = {
            desc = "DIPLOREASON_POSITIVE_OPINION"
            value = "scope:actor.opinion(scope:recipient)"
            multiply = 0.3
            min = 0       
        }
    
        if = {
            limit = {
                exists = situation:rise_of_the_ottomans.var:strongest_beylik_variable
                scope:actor ?= situation:rise_of_the_ottomans.var:strongest_beylik_variable
                exists = situation:rise_of_the_ottomans.var:second_strongest_strongest_beylik_variable
                scope:recipient ?= situation:rise_of_the_ottomans.var:second_strongest_strongest_beylik_variable
            }
            add = {
                value = -1000
                desc = "CHALLENGER_BEYLIK"
            }
        }

        if = {
            limit = {
                "scope:actor.opinion(scope:recipient)" < 150
                "scope:actor.conquer_desire(scope:recipient)" > 20
                scope:actor = {
                    is_neighbor_of = scope:recipient
                    NOT = {
                        wants_to_subjugate = scope:recipient
                    }
                    
                }
            }
            add = {
                desc = "RECIPIENTS_DESIRES_ACTORS_LOCATIONS"
                value = "scope:actor.conquer_desire(scope:recipient)"
                multiply = -0.5
                if = {
                    limit = {
                        scope:actor = {
                            ai_personality = ai_personality:ai_expansionist
                        }
                    }
                    add = {
                        desc = "DESIRES_LOCATIONS_TT_DESC"
                        value = "scope:actor.num_of_locations_with_high_conquer_desire(scope:recipient)"
                        multiply = -2.5
                    }
                }

            }
        }

        if = {
            #Recipient needs to be close and relativly big
            limit = {
                scope:actor = {
                    border_distance_to = {
                        country = scope:recipient
                        value < 150
                    }
                    relative_strength = {
                        target = scope:recipient
                        value < 0.60
                    }
                }
            }

            scope:actor = {         
                add = {
                    desc = "NEEDS_ALLIES_AGAINST_STRONG_RIVALS_AND_NEIGHBOURS"
                    # Check if the country needs allies at all
                    value = country_needs_allies_value
                    max = 25
                    min = 0
                }
            }
        }
        scope:recipient = {
            add = {
                desc = "COALITION_GRADE_ANTAGONISM_TOWARDS_RECIPIENT"
                value = {
                    every_country_with_coalition_grade_antagonism_against_us = {                    
                        limit = {
                            scope_country_can_join_coalition_against = {
                                target = scope:recipient
                            }
                        }
                        subtract = {
                            value = country_strength
                        }
                    }
                    if = {
                        limit = {
                            NOT = {
                                country_strength <= 0 
                            }
                        }
                        divide = country_strength
                    } 
                    multiply = 10
                }
            }
        }

        add = {
            desc = "ACTOR_WOULD_NEED_TO_PAY_FOR_DIPLOMACY"
            value = scope:actor.diplomatic_capacity_without_maintenance
            subtract = scope:actor.used_diplomatic_capacity
            if = {
                limit = {
                    NOT = { scope:actor.diplomatic_capacity_without_maintenance <= 0 }
                }
                divide = scope:actor.diplomatic_capacity_without_maintenance
            }
            max = 0
            multiply = 100
        }

        #Don't break alliances during wars
        if = {
            limit = {
                scope:actor = {
                    at_war = yes
                }
            }
            add = {
                desc = "ACTOR_IS_IN_DIFFERENT_WARS"
                value = 100
            }
        }

        add = {
            desc = "COMMON_RIVALS"
            value = "scope:recipient.num_of_common_rivals(scope:actor)"
            max = 3
            multiply = 5
        }

        # ---- v0.12 Geopolicy: the reasons of the region, milder for an alliance already made (the ally deciding =
        # scope:actor, its partner = scope:recipient) ----
        if = {
            limit = { NOT = { has_game_rule = gp_alliance_any } }
            if = {
                limit = {
                    scope:recipient = { gp_is_regional_rival_of = { COUNTRY = scope:actor } }
                    NOT = { gp_ally_overwhelming_threat = { WHO = scope:actor PARTNER = scope:recipient } }
                    NOT = { gp_ally_invader_front = { WHO = scope:actor PARTNER = scope:recipient } }	# v0.12.1 rivals of one region stand together against a foreign invader
                }
                add = {
                    desc = "GP_ALLY_REGION_RIVALS"
                    value = -30
                }
            }
            if = {
                limit = { scope:recipient = { has_variable = gp_sc_broken } }
                add = {
                    desc = "GP_ALLY_RUINED"
                    value = -40
                }
            }
            else_if = {
                limit = {
                    scope:recipient = { has_variable = gp_crisis var:gp_crisis >= 80 }
                    NOT = { gp_ally_invader_front = { WHO = scope:actor PARTNER = scope:recipient } }	# v0.12.1 beaten by the same invader: still a partner
                }
                add = {
                    desc = "GP_ALLY_RUINED"
                    value = -25
                }
            }
            else_if = {
                limit = {
                    scope:recipient = { has_variable = gp_recent_defeat }
                    NOT = { gp_ally_invader_front = { WHO = scope:actor PARTNER = scope:recipient } }
                }
                add = {
                    desc = "GP_ALLY_RUINED"
                    value = -10
                }
            }
            # v0.12.1 a foreign invader names both of us: the alliance is worth keeping (milder than making it, +40 against +60)
            if = {
                limit = { gp_ally_invader_front = { WHO = scope:actor PARTNER = scope:recipient } }
                add = {
                    desc = "GP_ALLY_INVADER"
                    value = 40
                }
            }
            if = {
                limit = {
                    gp_ally_far_without_cause = { WHO = scope:actor PARTNER = scope:recipient }
                    "scope:recipient.num_of_common_rivals(scope:actor)" < 1
                }
                add = {
                    desc = "GP_ALLY_FAR_NO_CAUSE"
                    value = -15
                }
            }
            # v0.12.1 an old friend is kept, milder than it is made (the ally deciding = scope:actor remembers the partner)
            if = {
                limit = { scope:recipient = { gp_is_old_friend_of = { COUNTRY = scope:actor } } }
                add = {
                    desc = "GP_ALLY_OLD_FRIEND"
                    value = 10
                }
            }
        }

		if = {
			limit = {
				exists = international_organization:protestant_union
				exists = international_organization:catholic_league
				OR = {
					AND = {
						scope:actor = { is_member_of_international_organization = international_organization:protestant_union }
						scope:recipient = { is_member_of_international_organization = international_organization:catholic_league }
					}
					AND = {
						scope:actor = { is_member_of_international_organization = international_organization:catholic_league }
						scope:recipient = { is_member_of_international_organization = international_organization:protestant_union }
					}
				}
			}
			subtract = {
				desc = "IN_OPPOSING_RELIGIOUS_FACTONS"
				value = 1000
			}
		}
	}
}
