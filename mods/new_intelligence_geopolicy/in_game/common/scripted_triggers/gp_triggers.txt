###################################
# New Intelligence: Geopolicy - scripted triggers (prefix gp_)
#
# No country modifiers anywhere (author's rule, 2026-11-04: no buffs, only behaviour): roles live in plain country
# variables (gp_role_*), temperaments are read on the fly from government / capital / faith.
# Strength comparisons: relative_strength is a SHARE 0..1 = our strength / (our + their), not a ratio
# (vanilla script_values comment "relative_strength(root) #this is 0..1"). Ratio R -> share R / (1 + R):
# 1.5x = 0.6, 2x = 0.67, 2.5x = 0.71, 3x = 0.75, 0.7x = 0.41, 0.35x = 0.26.
###################################

# Real countries only: no rebels, pirates, mercenary hosts, conquistador hosts, landless tags.
gp_is_real_country = {
	is_real_country = yes
	NOT = { is_subject_type = conquistador }
	num_locations > 0
}

# Game rule "Geopolicy AI": off disables roles, stages, threat reactions and scripted war scoring (defines stay).
gp_is_active = {
	NOT = { has_game_rule = gp_mode_off }
}

gp_is_strong_mode = {
	has_game_rule = gp_mode_strong
}

# ---- civilizational temperaments (read on the fly, no modifier) ----------------------------------

gp_is_steppe = {
	government_type = government_type:steppe_horde
}

gp_is_sinic = {
	exists = capital
	OR = {
		capital.region = region:north_china_region
		capital.region = region:east_china_region
		capital.region = region:south_china_region
		capital.region = region:west_china_region
	}
}

gp_is_japanese = {
	exists = capital
	capital.region = region:japan_region
}

gp_is_korean = {
	exists = capital
	capital.region = region:korea_region
}

gp_is_indian = {
	exists = capital
	capital.sub_continent = sub_continent:south_asia
}

gp_is_near_east_muslim = {
	exists = capital
	religion.group = religion_group:muslim
	OR = {
		capital.sub_continent = sub_continent:middle_east
		capital.sub_continent = sub_continent:north_africa
		capital.sub_continent = sub_continent:central_asia
	}
}

# ---- threat and intruders ----------------------------------------------------------------------

# this = candidate country, scope:gp_c = the country looking for its regional threat (saved in gp_yearly_update).
# The candidate must be at least 1.5x as strong (share >= gp_threat_share_min = 0.6).
gp_is_threat_candidate = {
	NOT = { this = scope:gp_c }
	gp_is_real_country = yes
	NOT = { is_subject_or_below_of = scope:gp_c }
	NOT = { scope:gp_c = { is_subject_or_below_of = prev } }
	NOT = { in_union_with = scope:gp_c }
	relative_strength = {
		target = scope:gp_c
		value >= gp_threat_share_min
	}
	OR = {
		is_threat_to = scope:gp_c
		is_neighbor_of = scope:gp_c
		any_owned_location = {
			count >= 5
			region = scope:gp_c.capital.region
		}
	}
}

# this = candidate, $COUNTRY$ = the country whose home region is intruded upon.
# An intruder keeps >= 3 locations in $COUNTRY$'s home region while its own capital sits in another region AND
# (another sub-continent OR another religion group). Overlord / subject / union partner never count.
# religion.group = X.religion.group: vanilla form ("NOT = { religion.group = root.religion.group }").
gp_is_intruder_of = {
	NOT = { this = $COUNTRY$ }
	gp_is_real_country = yes
	exists = capital
	$COUNTRY$ = { exists = capital }
	NOT = { is_subject_or_below_of = $COUNTRY$ }
	NOT = { $COUNTRY$ = { is_subject_or_below_of = prev } }
	NOT = { in_union_with = $COUNTRY$ }
	NOT = { capital.region = $COUNTRY$.capital.region }
	OR = {
		NOT = { capital.sub_continent = $COUNTRY$.capital.sub_continent }
		NOT = { religion.group = $COUNTRY$.religion.group }
	}
	any_owned_location = {
		count >= 3
		region = $COUNTRY$.capital.region
	}
}

# Context where root is the attacker (expansion target candidate lists).
gp_is_intruder = {
	gp_is_intruder_of = { COUNTRY = root }
}

# Both countries name the same regional threat (variable comparison with ?= so a missing variable is just "no").
gp_shares_threat_with = {
	has_variable = gp_threat
	var:gp_threat ?= $COUNTRY$.var:gp_threat
}

# ---- roles (plain country variables, set yearly by gp_update_role) --------------------------------

gp_is_expansion_role = {
	OR = {
		has_variable = gp_role_consolidator
		has_variable = gp_role_regional
		has_variable = gp_role_hegemon
		has_variable = gp_role_empire
	}
}
