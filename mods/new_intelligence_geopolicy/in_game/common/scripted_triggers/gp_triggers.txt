###################################
# New Intelligence: Geopolicy - scripted triggers (prefix gp_)
#
# No country modifiers anywhere (author's rule, 2026-11-04: no buffs, only behaviour): roles and postures live in
# plain country variables (gp_role_*, gp_posture_*), temperaments and the ruler are read on the fly.
# Strength comparisons: relative_strength is a SHARE 0..1 = our strength / (our + their), not a ratio
# (vanilla script_values comment "relative_strength(root) #this is 0..1"). Ratio R -> share R / (1 + R):
# 1.5x = 0.6, 2x = 0.67, 2.5x = 0.71, 3x = 0.75, 1x = 0.5, 0.8x = 0.45, 0.7x = 0.41, 0.35x = 0.26, 0.18x = 0.15.
###################################

# Real countries only: no rebels, pirates, mercenary hosts, conquistador hosts, landless tags.
gp_is_real_country = {
	is_real_country = yes
	NOT = { is_subject_type = conquistador }
	num_locations > 0
}

# Game rule "Geopolicy AI": off disables roles, postures, threat reactions and scripted war scoring (defines stay).
gp_is_active = {
	NOT = { has_game_rule = gp_mode_off }
}

gp_is_strong_mode = {
	has_game_rule = gp_mode_strong
}

# First twenty years of the campaign: countries mind their own region (author, 2026-11-04).
gp_is_early_game = {
	years_since_game_start < 20
}

# Who may turn inward: the Chinese, Japanese and Korean states and the Pacific islands (author: "Hawaii could be
# isolationist too"), and any country whose own laws and society shut it in (author: "if a country takes up something
# that gives it isolation") - the closed-borders immigration law or a strongly inward society (> 50, vanilla scale).
# v0.4 (author, 2026-11-06: "closedness only when access from outside is broken - countries always try to be active
# abroad"): no country is isolationist by birth any more (China, Japan, Korea and the Pacific islands included); only
# its own laws and society shut it in.
gp_isolation_home = {
	exists = capital
	OR = {
		has_policy = closed_borders_law
		societal_value:outward_vs_inward > 50
	}
}

# Isolation holds only while nothing forces the country out of it: no regional threat, no foreign power in the home
# region, no conquering ruler. When this turns false the posture is re-chosen at once (gp_update_posture).
gp_isolation_possible = {
	gp_isolation_home = yes
	NOT = { has_variable = gp_threat }
	NOT = { has_variable = gp_has_intruder }
	NOT = { ruler ?= { has_trait = conqueror } }
	NOT = { ruler ?= { mil >= 80 } }
}

# ---- threat and intruders ----------------------------------------------------------------------

# this = candidate country, scope:gp_c = the country looking for its regional threat (saved in gp_yearly_update).
# Two ways to become a threat: be at least 1.5x as strong (share 0.6) and border us / hold land in our region / be seen
# as a threat by the engine; or be EXPANDING (gp_is_expanding, momentum of conquest) and at least as strong as we are.
# The second branch makes neighbours of a fast conqueror close ranks even when it is not a giant.
gp_is_threat_candidate = {
	NOT = { this = scope:gp_c }
	gp_is_real_country = yes
	NOT = { is_subject_or_below_of = scope:gp_c }
	NOT = { scope:gp_c = { is_subject_or_below_of = prev } }
	NOT = { in_union_with = scope:gp_c }
	OR = {
		AND = {
			relative_strength = {
				target = scope:gp_c
				value >= gp_threat_share_min
			}
			OR = {
				is_neighbor_of = scope:gp_c
				any_owned_location = {
					count >= 5
					region = scope:gp_c.capital.region
				}
				# Across the sea the worry is halved (author: "the sea between England and France halves the unease"):
				# without a land border or a foothold in our region the engine's threat must be 3x as strong (share 0.75).
				AND = {
					is_threat_to = scope:gp_c
					relative_strength = {
						target = scope:gp_c
						value >= 0.75
					}
				}
			}
		}
		AND = {
			has_variable = gp_is_expanding
			relative_strength = {
				target = scope:gp_c
				value >= 0.5
			}
			OR = {
				is_neighbor_of = scope:gp_c
				any_owned_location = {
					count >= 3
					region = scope:gp_c.capital.region
				}
			}
		}
	}
}

# this = candidate, $COUNTRY$ = the country whose home region is intruded upon.
# An intruder keeps >= 3 locations in $COUNTRY$'s home region while its own capital sits in another REGION, and it is
# foreign (another sub-continent or another religion group) OR it is conquering right now (gp_is_expanding): England
# taking Normandy is an intruder for the French region while the war of conquest lasts. Overlord / subject / union
# partner never count. religion.group = X.religion.group: vanilla form.
gp_is_intruder_of = {
	NOT = { this = $COUNTRY$ }
	gp_is_real_country = yes
	exists = capital
	$COUNTRY$ = { exists = capital }
	NOT = { is_subject_or_below_of = $COUNTRY$ }
	NOT = { $COUNTRY$ = { is_subject_or_below_of = prev } }
	NOT = { in_union_with = $COUNTRY$ }
	NOT = { capital.region = $COUNTRY$.capital.region }
	OR = {
		NOT = { capital.sub_continent = $COUNTRY$.capital.sub_continent }
		NOT = { religion.group = $COUNTRY$.religion.group }
		has_variable = gp_is_expanding
	}
	any_owned_location = {
		count >= 3
		region = $COUNTRY$.capital.region
	}
}

# Context where root is the attacker (expansion target candidate lists).
gp_is_intruder = {
	gp_is_intruder_of = { COUNTRY = root }
}

# Both countries name the same regional threat (variable comparison with ?= so a missing variable is just "no").
gp_shares_threat_with = {
	has_variable = gp_threat
	var:gp_threat ?= $COUNTRY$.var:gp_threat
}

# ---- v0.12 the reasons of an alliance (scripted_relations/gp_alliance.txt; WHO decides, PARTNER is the other side) ----
# Quoted script values cannot take $PARAM$s (they are not substituted), so the common-rivals test stays in gp_alliance.txt.

# WHO and PARTNER name the same regional threat and it is overwhelming (3x, share 0.75): even rivals make common cause.
gp_ally_overwhelming_threat = {
	$WHO$ = {
		gp_shares_threat_with = { COUNTRY = $PARTNER$ }
		var:gp_threat_share >= 0.75
	}
}

# PARTNER lives in another region and nothing binds the two: no shared regional threat, and PARTNER is not at war with
# WHO's threat (author, 1628 review: "alliances outside the region only against a common enemy").
gp_ally_far_without_cause = {
	$WHO$ = { exists = capital }
	$PARTNER$ = { exists = capital }
	NOT = { $WHO$ = { capital.region = $PARTNER$.capital.region } }
	NOT = { $WHO$ = { gp_shares_threat_with = { COUNTRY = $PARTNER$ } } }
	NOT = { $WHO$.var:gp_threat ?= { is_at_war_with = $PARTNER$ } }
}

# A country of the western alliance game: France, Iberia, Britain, Ireland.
gp_is_western_game_country = {
	exists = capital
	OR = {
		capital.region = region:france_region
		capital.region = region:iberia_region
		capital.region = region:great_britain_region
		capital.region = region:ireland_region
	}
}

# Exactly one of the two plays the western game (author, 1628 review: "early in the game the alliance game is played in
# Western Europe; Austria, the German lands and Italy do not meddle in it") - read before 1450 only.
gp_ally_crosses_western_game = {
	OR = {
		AND = {
			$WHO$ = { gp_is_western_game_country = yes }
			$PARTNER$ = { gp_is_western_game_country = no }
		}
		AND = {
			$WHO$ = { gp_is_western_game_country = no }
			$PARTNER$ = { gp_is_western_game_country = yes }
		}
	}
}

# this = candidate partner of $COUNTRY$ against $COUNTRY$'s regional threat: same faith group, not the threat itself,
# and either it names the same threat or it is already at war with it (France fighting England for Brittany).
gp_is_front_partner_of = {
	NOT = { this = $COUNTRY$ }
	$COUNTRY$ = { has_variable = gp_threat }
	religion.group = $COUNTRY$.religion.group
	NOT = { this ?= $COUNTRY$.var:gp_threat }
	# v0.3: contenders of the same region stay rivals unless the threat is overwhelming (3x, share 0.75) - Novgorod and
	# Smolensk do not make friends just because Moscow grows (author, 1400 test).
	OR = {
		NOT = { gp_is_regional_rival_of = { COUNTRY = $COUNTRY$ } }
		$COUNTRY$ = { var:gp_threat_share >= 0.75 }
	}
	OR = {
		gp_shares_threat_with = { COUNTRY = $COUNTRY$ }
		$COUNTRY$.var:gp_threat ?= { is_at_war_with = prev }
	}
}

# Our threat is worth containing only while it is actually pressing on the region (conquering, or a foreign power in
# it) - static big neighbours are not dog-piled.
gp_threat_is_pressing = {
	has_variable = gp_threat
	OR = {
		var:gp_threat ?= { has_variable = gp_is_expanding }
		var:gp_threat ?= { gp_is_intruder_of = { COUNTRY = root } }
	}
}

# ---- v0.3 the region as the arena (gp_region_effects.txt) --------------------------------------------

# Until 1600 the home region is the whole game for all but hegemons and empires (author, 2026-11-05).
gp_is_before_1600 = {
	current_date < 1600.1.1
}

# Regions that history never united by war (the German lands of the Empire) get no unification casus belli.
gp_region_is_unifiable = {
	exists = capital
	OR = {
		AND = {
			NOT = { capital.region = region:north_german_region }
			NOT = { capital.region = region:south_german_region }
		}
		# v0.6 scenarios E-3 and E-5: the gatherer of Germany, or the Empire is no more
		has_variable = gp_sc_germany_gatherer
		has_global_variable = gp_sc_empire_gone
	}
}

# this and $COUNTRY$ are contenders of the same home region (Novgorod - Moscow - Smolensk).
gp_is_regional_rival_of = {
	NOT = { this = $COUNTRY$ }
	has_variable = gp_contender
	$COUNTRY$ = { has_variable = gp_contender }
	exists = capital
	$COUNTRY$ = { exists = capital }
	capital.region = $COUNTRY$.capital.region
	NOT = { is_subject_or_below_of = $COUNTRY$ }
	NOT = { $COUNTRY$ = { is_subject_or_below_of = prev } }
}

# root may use gp_cb_regional_unification: its region can be united by war, and either it holds 90 % of it and the claim
# came up (gp_region_claim), or it is a contender and the region is in tension (ten years of peace or a time of troubles).
gp_has_unification_right = {
	gp_region_is_unifiable = yes
	OR = {
		has_variable = gp_region_claim
		AND = {
			has_variable = gp_contender
			OR = {
				has_variable = gp_region_turmoil
				AND = {
					has_variable = gp_calm_years
					var:gp_calm_years >= 10
				}
			}
		}
	}
}
