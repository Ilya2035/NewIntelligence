# New Intelligence: Geopolicy - new formable countries (prefix gp_)

# Format: formable_countries/readme.txt and 00_formable_countries.txt (level, required_locations_fraction, rule,
# potential / allow, name / flag / adjective / tag / color, regions / areas / provinces, form_effect).
# Every tag here is NEW (checked against setup/countries, coat_of_arms and country_names loc): GAE AMC AGV WND BGD
# ILY DAC GVO NVR JRS KHO. Colors: main_menu/common/named_colors/gp_colors.txt; arms: coat_of_arms/gp_formable_coa.txt;
# names: gp_geopolicy_l_*.yml (<TAG>, <TAG>_ADJ, <key>, <key>_desc).
# rule = historical / plausible is honoured by the vanilla game rule "Ahistorical formable countries".
# Each formable keeps a one-line historical hook in its comment - the "if it had happened" the mod is built on.

# Alba - the pan-Gaelic kingdom. 1315-18 Edward Bruce was crowned High King of Ireland while his brother Robert
# wrote to the Irish kings of "our nation and the common language"; here the Bruce project succeeds.
gp_alba_f = {
	level = 3
	required_locations_fraction = 0.6
	rule = plausible
	content_priority = 700

	potential = {
		OR = {
			culture = { has_culture_group = culture_group:scottish_group }
			culture = { has_culture_group = culture_group:hibernian_group }
		}
	}

	allow = {
		owns = location:edinburgh
		any_owned_location = { region = region:ireland_region }
	}

	name = GAE
	flag = GAE
	adjective = GAE_ADJ
	tag = GAE
	color = map_GAE

	regions = {
		ireland_region
	}
	areas = {
		scottish_lowlands_area
		scottish_highlands_area
		hebrides_area
	}
	form_effect = {
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
	}
}
