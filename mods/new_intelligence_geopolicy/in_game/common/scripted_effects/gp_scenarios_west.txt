# New Intelligence: Geopolicy - scenarios of the West (prefix gp_), v0.5
#
# docs/GEOPOLICY_SCENARIOS_WEST.md, all branches chosen by the author (2026-11-06). A branch opens once, from the outcome
# of wars (who holds what), never from a date alone, and stays open. Global variables:
#   gp_sc_hyw       the Hundred Years' War, v0.8 state machine: gp_scenarios_hyw.txt (codes 1-8 are listed there)
#   gp_sc_ireland   1 Ireland united, 2 the Anglo-Irish (quick English conquest), 3 Irish national consciousness
#   gp_sc_ireland_quarrel   the Gaelic lords quarrelled (by 1400 nobody holds 30 % of Ireland)
#   gp_sc_scotland  1 the Auld Alliance holds (default), 2 Scotland wins Britain
#   gp_sc_reconquista, gp_sc_portugal, gp_sc_aragon_sea, gp_sc_granada_holds   Iberia (independent flags)
#   gp_sc_france_leader   the gatherer of France in states 4 and 5 (a country)
# scope:gp_eng / scope:gp_fra are the English and French crowns (v0.8, gp_sc_save_crowns), not simply the owners of
# London and Paris.
# Checked once a year by the owner of London (gp_yearly_update), so exactly one country runs it. Effects act through
# postures, war targets, the scenario casus belli (gp_cb_scenario_war), patrons, objectives and direction preferences;
# the two Irish location modifiers obey the game rule "Scenario marks on the land" (gp_rule_scenario_mods).
###################################

# ---- helpers -----------------------------------------------------------------------------------------

# Share of region $REGION$ owned by scope:$WHO$, into variable $VAR$ of scope:gp_c (the anchor).
gp_sc_share = {
	set_variable = { name = gp_sc_tmp_total value = 0 }
	set_variable = { name = gp_sc_tmp_owned value = 0 }
	region:$REGION$ = {
		every_ownable_location_in_region = {
			scope:gp_c = { change_variable = { name = gp_sc_tmp_total add = 1 } }
			if = {
				limit = { owner ?= scope:$WHO$ }
				scope:gp_c = { change_variable = { name = gp_sc_tmp_owned add = 1 } }
			}
		}
	}
	set_variable = {
		name = $VAR$
		value = {
			value = var:gp_sc_tmp_owned
			divide = {
				value = var:gp_sc_tmp_total
				min = 1
			}
		}
	}
	remove_variable = gp_sc_tmp_total
	remove_variable = gp_sc_tmp_owned
}

# Country scope: an AI country prepares a war against scope:$TARGET$ for six years and may use the scenario casus belli.
gp_sc_prepare = {
	set_variable = { name = gp_sc_cb_target value = scope:$TARGET$ }
	if = {
		limit = {
			is_ai = yes
			gp_is_active = yes
			NOT = { this = scope:$TARGET$ }
		}
		gp_set_posture = { POSTURE = preparation }
		set_variable = { name = gp_war_target value = scope:$TARGET$ }
		set_variable = { name = gp_posture_left value = 6 }
	}
}

# Country scope: an AI country is set on a course for $YEARS$ years.
gp_sc_course = {
	if = {
		limit = {
			is_ai = yes
			gp_is_active = yes
		}
		gp_set_posture = { POSTURE = $POSTURE$ }
		set_variable = { name = gp_posture_left value = $YEARS$ }
		if = { limit = { has_variable = gp_war_target } remove_variable = gp_war_target }
	}
}

# Country scope: broken for $YEARS$ years - counted as a hopeless minor (gp_update_contender), whatever its size.
gp_sc_break = {
	set_variable = gp_sc_broken
	set_variable = { name = gp_sc_broken_years value = $YEARS$ }
}

# Every country, yearly (gp_yearly_update): counters of the scenario states.
gp_sc_country_upkeep = {
	if = {
		limit = { has_variable = gp_sc_broken_years }
		change_variable = { name = gp_sc_broken_years add = -1 }
		if = {
			limit = { var:gp_sc_broken_years <= 0 }
			remove_variable = gp_sc_broken_years
			if = { limit = { has_variable = gp_sc_broken } remove_variable = gp_sc_broken }
		}
	}
	# 3A: Scotland never yields to England while the Auld Alliance branch holds and Scotland is free.
	if = { limit = { has_variable = gp_sc_never_yield } remove_variable = gp_sc_never_yield }
	if = {
		limit = {
			tag = SCO
			is_subject = no
			has_global_variable = gp_sc_scotland
			global_var:gp_sc_scotland = 1
		}
		set_variable = gp_sc_never_yield
	}
	if = {
		limit = {
			has_variable = gp_sc_cb_target
			NOT = { var:gp_sc_cb_target ?= { gp_is_real_country = yes } }
		}
		remove_variable = gp_sc_cb_target
	}
	gp_sc_hyw_country_upkeep = yes	# v0.8 colony bans, Scotland's hour, marks of lost guarantees (gp_scenarios_hyw.txt)
	gp_ib_country_upkeep = yes	# v0.9 the Pyrenees: a foreign crown never settles (gp_scenarios_iberia.txt)
}

# ================================================================================================
# Yearly check (country scope = the owner of London, scope:gp_c)
# ================================================================================================

gp_west_scenarios = {
	gp_sc_save_crowns = yes	# v0.8: scope:gp_eng / scope:gp_fra = the crowns (gp_scenarios_hyw.txt)
	if = {
		limit = {
			exists = scope:gp_eng
			exists = scope:gp_fra
		}
		gp_sc_hundred_years_war = yes
		gp_sc_ireland = yes
		gp_sc_scotland = yes
	}
	gp_sc_iberia = yes
	gp_sc_france_upkeep = yes
	gp_central_scenarios = yes	# v0.6 Italy and the Empire (gp_scenarios_central.txt)
	clear_saved_scope = gp_eng
	clear_saved_scope = gp_fra
}

# ---- 1. The Hundred Years' War: gp_scenarios_hyw.txt (v0.8) ------------------------------------------

# ---- 2. Ireland -------------------------------------------------------------------------------------

gp_sc_ireland = {
	if = {
		limit = { NOT = { has_global_variable = gp_sc_ireland } }
		# the strongest Gaelic country of Ireland
		region:ireland_region = {
			ordered_present_country = {
				limit = {
					gp_is_real_country = yes
					exists = capital
					capital.region = region:ireland_region
					culture ?= { has_culture_group = culture_group:hibernian_group }
				}
				order_by = country_strength
				max = 1
				save_scope_as = gp_gael
			}
		}
		gp_sc_share = { REGION = ireland_region WHO = gp_eng VAR = gp_sc_eng_in_ireland }
		# 2A Ireland united by a Gaelic king (half of the island before England)
		if = {
			limit = {
				scope:gp_gael ?= {
					has_variable = gp_region_share
					var:gp_region_share >= 0.5
				}
			}
			set_global_variable = { name = gp_sc_ireland value = 1 }
			scope:gp_gael = {
				c:SCO ?= {
					add_opinion = { target = scope:gp_gael modifier = gp_opinion_ancient_friendship }
					reverse_add_opinion = { target = scope:gp_gael modifier = gp_opinion_ancient_friendship }
				}
				scope:gp_fra = {
					add_opinion = { target = scope:gp_gael modifier = gp_opinion_ancient_friendship }
					reverse_add_opinion = { target = scope:gp_gael modifier = gp_opinion_ancient_friendship }
				}
				if = {
					limit = { NOT = { has_game_rule = gp_scenario_mods_off } }
					every_owned_location = {
						limit = { region = region:ireland_region }
						add_location_modifier = { modifier = gp_sc_irish_resolve days = 18250 }
					}
				}
			}
		}
		# 2B / 2C England holds three quarters of Ireland - early: the Anglo-Irish; late: national consciousness
		else_if = {
			limit = { var:gp_sc_eng_in_ireland >= 0.75 }
			if = {
				limit = { current_date < 1450.1.1 }
				set_global_variable = { name = gp_sc_ireland value = 2 }
			}
			else = {
				set_global_variable = { name = gp_sc_ireland value = 3 }
				region:ireland_region = {
					every_present_country = {
						limit = {
							gp_is_real_country = yes
							culture ?= { has_culture_group = culture_group:hibernian_group }
						}
						set_variable = { name = gp_sc_cb_target value = scope:gp_eng }
					}
				}
			}
		}
		remove_variable = gp_sc_eng_in_ireland
		clear_saved_scope = gp_gael
	}
	# 2D the Gaelic lords quarrelled: by 1400 nobody holds 30 % of Ireland - England's road there opens early
	if = {
		limit = {
			NOT = { has_global_variable = gp_sc_ireland_quarrel }
			NOT = { has_global_variable = gp_sc_ireland }
			current_date >= 1400.1.1
			NOT = {
				region:ireland_region = {
					any_present_country = {
						exists = capital
						capital.region = region:ireland_region
						has_variable = gp_region_share
						var:gp_region_share >= 0.3
					}
				}
			}
		}
		set_global_variable = { name = gp_sc_ireland_quarrel value = 1 }
		scope:gp_eng = { set_variable = gp_sc_ireland_right }
	}
}

# ---- 3. Scotland --------------------------------------------------------------------------------

gp_sc_scotland = {
	if = {
		limit = {
			NOT = { has_global_variable = gp_sc_scotland }
			c:SCO ?= { is_subject = no }
		}
		set_global_variable = { name = gp_sc_scotland value = 1 }
	}
	# 3B Scotland wins Britain (40 % of the region)
	if = {
		limit = {
			has_global_variable = gp_sc_scotland
			global_var:gp_sc_scotland = 1
			c:SCO ?= {
				has_variable = gp_region_share
				var:gp_region_share >= 0.4
			}
		}
		set_global_variable = { name = gp_sc_scotland value = 2 }
		scope:gp_eng = {
			if = {
				limit = { NOT = { tag = SCO } }
				gp_sc_break = { YEARS = 15 }
			}
		}
	}
	# the French crown stands by Scotland while the Auld Alliance holds
	if = {
		limit = {
			has_global_variable = gp_sc_scotland
			global_var:gp_sc_scotland = 1
			NOT = { scope:gp_fra = scope:gp_eng }
		}
		scope:gp_fra = {
			c:SCO ?= { save_scope_as = gp_sco }
			if = {
				limit = { exists = scope:gp_sco }
				set_variable = { name = gp_sc_protect value = scope:gp_sco }
			}
			clear_saved_scope = gp_sco
		}
	}
}

# ---- 4. Iberia: the Iberian chess game, gp_scenarios_iberia.txt (v0.9) ---------------------------------

# ================================================================================================
# Directions set by the scenarios (called at the end of gp_update_directions, AI only, scope:gp_c)
# ================================================================================================

gp_sc_directions = {
	# 1A England turns to its islands (the island power, v0.8), France to Italy
	if = {
		limit = { has_variable = gp_sc_island_power }
		gp_dir_drop = { DIR = british }
		gp_sc_dir_add = { DIR = british_isles }
	}
	if = {
		limit = {
			has_global_variable = gp_sc_hyw
			OR = {
				global_var:gp_sc_hyw = 1
				global_var:gp_sc_hyw = 2
			}
			global_var:gp_sc_fra_crown ?= scope:gp_c
		}
		gp_sc_dir_add = { DIR = french }
	}
	# 1C / 1D Burgundy reaches for the Low Countries and Italy
	if = {
		limit = {
			tag = BUR
			has_global_variable = gp_sc_hyw
			OR = {
				global_var:gp_sc_hyw = 4
				global_var:gp_sc_hyw = 5
			}
		}
		gp_sc_dir_add = { DIR = french }
	}
	# 3B victorious Scotland looks to Ireland
	if = {
		limit = {
			OR = {
				tag = SCO
				tag = GAE
			}
			OR = {
				AND = {
					has_global_variable = gp_sc_scotland
					global_var:gp_sc_scotland = 2
				}
				has_variable = gp_sc_scot_hour_years	# v0.8 Scotland's hour after the break-up of England
			}
		}
		gp_sc_dir_add = { DIR = british_isles }
	}
	# 4A Castile crosses to Africa
	if = {
		limit = {
			has_global_variable = gp_sc_reconquista
			OR = {
				tag = CAS
				tag = SPA
			}
		}
		gp_sc_dir_add = { DIR = iberian }
	}
	# 4C Aragon in Italy and the Maghreb
	if = {
		limit = {
			has_global_variable = gp_sc_aragon_sea
			OR = {
				tag = ARA
				tag = MDC
			}
		}
		gp_sc_dir_add = { DIR = italian }
	}
	# 4D Morocco looks to Iberia
	if = {
		limit = {
			has_global_variable = gp_sc_granada_holds
			tag = MOR
		}
		gp_sc_dir_add = { DIR = maghrebi }
	}
	# v0.9 Portugal, master of its Iberian game, crosses to the Maghreb (and no further into Africa)
	if = {
		limit = {
			has_variable = gp_ib_overseas
			global_var:gp_ib_por ?= scope:gp_c
		}
		gp_sc_dir_add = { DIR = maghrebi }
	}
	gp_sc_directions_central = yes	# v0.6 (gp_scenarios_central.txt)
}

gp_sc_dir_add = {
	if = {
		limit = { NOT = { has_area_preference = gp_dir_$DIR$ } }
		add_area_preference = gp_dir_$DIR$
	}
}

# ================================================================================================
# Marks on the land (called from gp_control_location, location scope, owner = scope:gp_c)
# ================================================================================================

# 2C Irish national consciousness: Irish land under a non-Gaelic owner -15 % max control, for good.
gp_sc_control_location = {
	if = {
		limit = {
			NOT = { has_game_rule = gp_scenario_mods_off }
			has_global_variable = gp_sc_ireland
			global_var:gp_sc_ireland = 3
			region = region:ireland_region
			dominant_culture ?= { has_culture_group = culture_group:hibernian_group }
			NOT = { scope:gp_c = { culture ?= { has_culture_group = culture_group:hibernian_group } } }
		}
		if = {
			limit = { NOT = { has_location_modifier = gp_sc_irish_nation } }
			add_location_modifier = { modifier = gp_sc_irish_nation days = -1 }
		}
	}
	else_if = {
		limit = { has_location_modifier = gp_sc_irish_nation }
		remove_location_modifier = gp_sc_irish_nation
	}
}
