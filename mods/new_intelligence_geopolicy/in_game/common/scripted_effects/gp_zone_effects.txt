###################################
# New Intelligence: Geopolicy - zone profiles by tag (prefix gp_)
#
# The preferences themselves live in in_game/common/area_preferences/gp_zones.txt. Four switches map a tag to its
# core / sphere / far preference (switch = { trigger = tag ... }: vanilla scripted_effects use switch with
# trigger = $location$ and location:x = { } branches; with trigger = tag the branch keys are bare tags).
# Countries without a tag profile fall back to gp_generic_home_region (unless vanilla already gave them
# generic_conquer_home_region) and to the group spheres (Anatolian beyliks, Persian emirates) by trigger.
# A formable that switches tag (BRA -> PRU, ENG -> GBR) re-runs gp_assign_zone_profile via on_new_country_formed;
# the old core preference stays (same areas, harmless).
###################################

gp_add_pref = {
	if = {
		limit = { NOT = { has_area_preference = $PREF$ } }
		add_area_preference = $PREF$
	}
}

gp_remove_pref = {
	if = {
		limit = { has_area_preference = $PREF$ }
		remove_area_preference = $PREF$
	}
}
