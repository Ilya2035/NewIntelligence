# Contender or minor (yearly, before the region relations)
# ================================================================================================

# Contender: a historical contender, or a country holding a quarter of its region, or - when no historical contender is
# left in the region - its strongest independent country (the Eretnids take the Ottomans' place only if the Ottomans are
# gone). Hopeless: not a contender, little of the region, and a country at least twice as strong has its capital in the
# region - such a country defends and looks for friends instead of carving a sausage-shaped realm.
gp_update_contender = {
	if = { limit = { has_variable = gp_contender } remove_variable = gp_contender }
	if = { limit = { has_variable = gp_hopeless } remove_variable = gp_hopeless }
	if = {
		limit = {
			exists = capital
			is_subject = no
		}
		if = {
			limit = {
				NOT = { has_variable = gp_sc_broken }	# v0.5 a broken power (scenario) is no contender
				OR = {
					has_variable = gp_historic_contender
					var:gp_region_share >= 0.25
					AND = {
						NOT = {
							capital.region = {
								any_present_country = {
									has_variable = gp_historic_contender
									is_subject = no
									exists = capital
									capital.region = scope:gp_c.capital.region
								}
							}
						}
						NOT = {
							capital.region = {
								any_present_country = {
									NOT = { this = scope:gp_c }
									gp_is_real_country = yes
									is_subject = no
									exists = capital
									capital.region = scope:gp_c.capital.region
									country_strength > scope:gp_c.country_strength
								}
							}
						}
					}
				}
			}
			set_variable = gp_contender
		}
		else_if = {
			limit = { has_variable = gp_sc_broken }
			set_variable = gp_hopeless
		}
		else_if = {
			limit = {
				var:gp_region_share < 0.1
				capital.region = {
					any_present_country = {
						NOT = { this = scope:gp_c }
						gp_is_real_country = yes
						exists = capital
						capital.region = scope:gp_c.capital.region
						relative_strength = {
							target = scope:gp_c
							value >= 0.67
						}
					}
				}
			}
			set_variable = gp_hopeless
		}
	}
}

# ================================================================================================
# The country's mentality follows its course (author: "not 'balanced' everywhere - defense, status quo or conquest").
# Vanilla AI personalities (in_game/common/ai_personalities, effect set_personality as in ai_personalities_setup.txt).
# Called from gp_set_posture, so every change of course changes the mentality.
# ================================================================================================

gp_apply_personality = {
	if = {
		limit = {
			is_ai = yes
			gp_is_active = yes
		}
		if = {
			limit = { has_variable = gp_posture_preparation }
			set_personality = ai_personality:ai_aggressive
		}
		else_if = {
			limit = { has_variable = gp_posture_expansion }
			set_personality = ai_personality:ai_expansionist
		}
		else_if = {
			limit = { has_variable = gp_posture_opportunism }
			set_personality = ai_personality:ai_opportunistic
		}
		else_if = {
			limit = {
				OR = {
					has_variable = gp_posture_defense
					has_variable = gp_posture_common_front
				}
			}
			set_personality = ai_personality:ai_defensive
		}
		else_if = {
			limit = {
				OR = {
					has_variable = gp_posture_consolidation
					has_variable = gp_posture_recovery
				}
			}
			set_personality = ai_personality:ai_cautious
		}
		else_if = {
			limit = { has_variable = gp_posture_peace }
			set_personality = ai_personality:ai_friendly
		}
		else_if = {
			limit = { has_variable = gp_posture_isolation }
			set_personality = ai_personality:ai_isolationist
		}
	}
}

# ================================================================================================
# The region's own claim (author: "who holds 90 % of a region gets, with 10 % a year, a casus belli on the rest")
# ================================================================================================

gp_update_region_claim = {
	if = {
		limit = { has_variable = gp_region_claim_years }
		change_variable = { name = gp_region_claim_years add = -1 }
		if = {
			limit = { var:gp_region_claim_years <= 0 }
			remove_variable = gp_region_claim_years
			if = { limit = { has_variable = gp_region_claim } remove_variable = gp_region_claim }
		}
	}
	if = {
		limit = {
			var:gp_region_share >= 0.9
			NOT = { has_variable = gp_region_claim }
		}
		random_list = {
			10 = {
				set_variable = gp_region_claim
				set_variable = { name = gp_region_claim_years value = 10 }
			}
			90 = { }
		}
	}
}

# ================================================================================================
# Control of the land (game rule "Holding the land", location modifiers for every owner; requested by the author)
#   - another region of the home macro-region (sub-continent): -25 % max control until 1600;
#   - another macro-region: -50 % until 1600, -25 % until 1700; nothing after 1700;
#   - land of our culture group and faith counts as home (acclimatised), wherever it lies;
#   - a country that has just lost a war (gp_recent_defeat) loses its grip on the outer lands first: another -15 %;
#   - ragged borders: a location with no neighbour of ours (enclave) -50 %, a narrow salient (at most two neighbours of
#     ours and at least three foreign ones) -20 %.
# Static modifiers: main_menu/common/static_modifiers/gp_control_modifiers.txt. Recomputed yearly per owned location.
# ================================================================================================

gp_update_control = {
	if = {
		limit = {
			exists = capital
			NOT = { has_game_rule = gp_control_off }
		}
		every_owned_location = {
			gp_control_location = yes
		}
	}
}

# location scope, owner = scope:gp_c
gp_control_location = {
	# ---- distance from the home region ----
	if = {
		limit = {
			OR = {
				region = scope:gp_c.capital.region
				# v0.5 branch 2B: the Anglo-Irish - Ireland is home to the English crown
				AND = {
					region = region:ireland_region
					has_global_variable = gp_sc_ireland
					global_var:gp_sc_ireland = 2
					location:london.owner ?= scope:gp_c
				}
				AND = {
					dominant_culture ?= { has_shared_culture_group = scope:gp_c.culture }
					dominant_religion ?= scope:gp_c.religion
				}
			}
		}
		gp_control_set_distance = { LEVEL = none }
	}
	else_if = {
		limit = { sub_continent = scope:gp_c.capital.sub_continent }
		if = {
			limit = { current_date < 1600.1.1 }
			gp_control_set_distance = { LEVEL = 1 }
		}
		else = {
			gp_control_set_distance = { LEVEL = none }
		}
	}
	else = {
		if = {
			limit = { current_date < 1600.1.1 }
			gp_control_set_distance = { LEVEL = 2 }
		}
		else_if = {
			limit = { current_date < 1700.1.1 }
			gp_control_set_distance = { LEVEL = 1 }
		}
		else = {
			gp_control_set_distance = { LEVEL = none }
		}
	}
	# ---- the grip after a defeat (outer lands only) ----
	if = {
		limit = {
			scope:gp_c = { has_variable = gp_recent_defeat }
			NOT = { region = scope:gp_c.capital.region }
		}
		if = {
			limit = { NOT = { has_location_modifier = gp_control_shaken } }
			add_location_modifier = { modifier = gp_control_shaken days = -1 }
		}
	}
	else_if = {
		limit = { has_location_modifier = gp_control_shaken }
		remove_location_modifier = gp_control_shaken
	}
	# ---- v0.5 scenario marks (Irish national consciousness) ----
	gp_sc_control_location = yes
	# ---- the shape of the border ----
	if = {
		limit = {
			NOT = { any_neighbor_location = { owner ?= scope:gp_c } }
		}
		gp_control_set_shape = { LEVEL = enclave }
	}
	else_if = {
		limit = {
			NOT = {
				any_neighbor_location = {
					count >= 3
					owner ?= scope:gp_c
				}
			}
			any_neighbor_location = {
				count >= 3
				exists = owner
				NOT = { owner ?= scope:gp_c }
			}
		}
		gp_control_set_shape = { LEVEL = salient }
	}
	else = {
		gp_control_set_shape = { LEVEL = none }
	}
}

# Keeps exactly one of the distance modifiers (or none). LEVEL = none / 1 / 2.
gp_control_set_distance = {
	if = {
		limit = { has_location_modifier = gp_control_far_1 }
		remove_location_modifier = gp_control_far_1
	}
	if = {
		limit = { has_location_modifier = gp_control_far_2 }
		remove_location_modifier = gp_control_far_2
	}
	gp_control_add_far_$LEVEL$ = yes
}

gp_control_add_far_none = { }
gp_control_add_far_1 = {
	add_location_modifier = { modifier = gp_control_far_1 days = -1 }
}
gp_control_add_far_2 = {
	add_location_modifier = { modifier = gp_control_far_2 days = -1 }
}

# LEVEL = none / enclave / salient.
gp_control_set_shape = {
	if = {
		limit = { has_location_modifier = gp_control_enclave }
		remove_location_modifier = gp_control_enclave
	}
	if = {
		limit = { has_location_modifier = gp_control_salient }
		remove_location_modifier = gp_control_salient
	}
	gp_control_add_shape_$LEVEL$ = yes
}

gp_control_add_shape_none = { }
gp_control_add_shape_enclave = {
	add_location_modifier = { modifier = gp_control_enclave days = -1 }
}
gp_control_add_shape_salient = {
	add_location_modifier = { modifier = gp_control_salient days = -1 }
}

# ================================================================================================
# "Balance of power" tab of the Geopolitics window (in_game/gui/gp_balance_tab.gui), players only.
# Country scope, scope:gp_c = the player. Lists (variable lists of countries, shown with MakeScope.GetList as vanilla
# treasure_fleet "visited_countries"):
#   gp_bp_contenders - the five strongest independent countries with their capital in our home region
#   gp_bp_outsiders  - the four strongest countries from outside holding at least three locations in it
#   gp_bp_macro      - the six strongest independent countries of our macro-region (sub-continent)
# and var:gp_bp_holder - the country of our region holding the largest share of it.
# Refreshed yearly (gp_yearly_update) and by the tab's refresh button (gp_scripted_guis.txt).
# ================================================================================================

gp_update_balance = {
	if = { limit = { has_variable_list = gp_bp_contenders } clear_variable_list = gp_bp_contenders }
	if = { limit = { has_variable_list = gp_bp_outsiders } clear_variable_list = gp_bp_outsiders }
	if = { limit = { has_variable_list = gp_bp_macro } clear_variable_list = gp_bp_macro }
	if = { limit = { has_variable = gp_bp_holder } remove_variable = gp_bp_holder }
	if = {
		limit = { exists = capital }
		capital.region = {
			ordered_present_country = {
				limit = {
					gp_is_real_country = yes
					is_subject = no
					exists = capital
					capital.region = scope:gp_c.capital.region
				}
				order_by = country_strength
				max = 5
				scope:gp_c = { add_to_variable_list = { name = gp_bp_contenders target = prev } }
			}
			ordered_present_country = {
				limit = {
					gp_is_real_country = yes
					exists = capital
					NOT = { capital.region = scope:gp_c.capital.region }
					any_owned_location = {
						count >= 3
						region = scope:gp_c.capital.region
					}
				}
				order_by = country_strength
				max = 4
				scope:gp_c = { add_to_variable_list = { name = gp_bp_outsiders target = prev } }
			}
			ordered_present_country = {
				limit = {
					gp_is_real_country = yes
					exists = capital
					capital.region = scope:gp_c.capital.region
				}
				order_by = gp_region_share_or_zero
				max = 1
				scope:gp_c = { set_variable = { name = gp_bp_holder value = prev } }
			}
		}
		ordered_country = {
			limit = {
				gp_is_real_country = yes
				is_subject = no
				exists = capital
				capital.sub_continent = scope:gp_c.capital.sub_continent
			}
			order_by = country_strength
			max = 6
			scope:gp_c = { add_to_variable_list = { name = gp_bp_macro target = prev } }
		}
	}
}

# The tab's refresh button: today's region share and lists, without the yearly side effects (momentum, calm years).
gp_refresh_balance = {
	save_scope_as = gp_c
	gp_compute_region_share = yes
	gp_update_balance = yes
	clear_saved_scope = gp_c
}

# Overseas: a contender holding 40 % of its home region (or anyone holding 60 %) of the colonising regions (author:
# France, Britain, Ireland, Iberia, the Maghreb, Italy, the Balkans, Anatolia, the German lands, Scandinavia) loses the
# vanilla "avoid_*_colonization" preferences (setup_area_preferences gives them to every AI at game start) and gets the
# explorer's taste for the New World; Russia opens Siberia, India and China open South-East Asia. Losing the position
# restores the vanilla preferences. AI only.
gp_update_overseas = {
	if = {
		limit = {
			exists = capital
			is_subject = no
			OR = {
				var:gp_region_share >= 0.6
				AND = {
					has_variable = gp_contender
					var:gp_region_share >= 0.4
				}
			}
		}
		if = {
			limit = {
				OR = {
					capital.sub_continent = sub_continent:western_europe
					capital.region = region:maghreb_region
					capital.region = region:balkan_region
					capital.region = region:anatolia_region
				}
			}
			gp_pref_drop = { PREF = avoid_american_colonization }
			gp_pref_drop = { PREF = avoid_african_colonization }
			gp_pref_drop = { PREF = avoid_south_east_asica_colonization }
			if = {
				limit = { NOT = { has_area_preference = gp_colonial_ambition } }
				add_area_preference = gp_colonial_ambition
			}
			set_variable = gp_overseas_open
		}
		if = {
			limit = { capital.region = region:russian_region }
			gp_pref_drop = { PREF = avoid_siberia_colonization }
			set_variable = gp_overseas_open
		}
		if = {
			limit = {
				OR = {
					capital.sub_continent = sub_continent:south_asia
					gp_is_sinic = yes
				}
			}
			gp_pref_drop = { PREF = avoid_south_east_asica_colonization }
			set_variable = gp_overseas_open
		}
	}
	else_if = {
		limit = { has_variable = gp_overseas_open }
		remove_variable = gp_overseas_open
		gp_pref_drop = { PREF = gp_colonial_ambition }
		if = {
			limit = { exists = capital }
			if = {
				limit = { NOT = { capital.continent = continent:america } }
				gp_pref_keep = { PREF = avoid_american_colonization }
			}
			if = {
				limit = { NOT = { capital.continent = continent:africa } }
				gp_pref_keep = { PREF = avoid_african_colonization }
			}
			if = {
				limit = { NOT = { capital.sub_continent = sub_continent:south_east_asia } }
				gp_pref_keep = { PREF = avoid_south_east_asica_colonization }
			}
			gp_pref_keep = { PREF = avoid_siberia_colonization }
		}
	}
}

gp_pref_drop = {
	if = {
		limit = { has_area_preference = $PREF$ }
		remove_area_preference = $PREF$
	}
}

gp_pref_keep = {
	if = {
		limit = { NOT = { has_area_preference = $PREF$ } }
		add_area_preference = $PREF$
	}
}

# No eternal allies (author: "there are no eternal allies and no eternal enemies, especially late in the game"): from
# 1450 every year of an alliance that no common threat holds together cools our opinion of the ally a little; the
# memory fades three times faster than it grows once the alliance is gone (gp_opinion_alliance_fatigue). Stronger after
# 1600 and 1700 (gp_alliance_fatigue_scale). Old enmities already fade by themselves (gp_opinions.txt).
gp_update_alliance_fatigue = {
	if = {
		limit = { current_date >= 1450.1.1 }
		every_ally = {
			limit = {
				gp_is_real_country = yes
				NOT = { is_subject_or_below_of = scope:gp_c }
				NOT = { scope:gp_c = { is_subject_or_below_of = prev } }
				NOT = { gp_shares_threat_with = { COUNTRY = scope:gp_c } }
				NOT = { this ?= scope:gp_c.var:gp_ruler_like }
			}
			scope:gp_c = {
				add_opinion = { target = prev modifier = gp_opinion_alliance_fatigue scale = gp_alliance_fatigue_scale }
			}
		}
	}
}
