# New Intelligence: Geopolicy - the Hundred Years' War and the break-up of the crowns (prefix gp_), v0.8
#
# The author's rework of 2026-11-06 (docs/GEOPOLICY_SCENARIOS_WEST.md, section 8). One state machine, checked once a
# year by the owner of London (gp_west_scenarios). Global variables:
#   gp_sc_eng_crown / gp_sc_fra_crown  the English and the French crown: the owners of London and Paris at the first
#                     check (ENG, FRA). While a crown lives it stays the crown, wherever its capital goes; a dead crown is
#                     replaced by the owner of London / Paris (scope:gp_eng / scope:gp_fra, gp_sc_save_crowns).
#   gp_sc_hyw         none  the war for the crown (1337-1400)
#                     3     English advantage: England (with its vassals) holds half of Brittany, Normandy and Picardy
#                           before 1400 - Crecy, and France must deal with it by 1400
#                     5     dual monarchy: England holds Paris - France has 30 years to free it
#                     4     the break-up of France: Paris not freed in 30 years - France falls into Brittany, Normandy,
#                           Burgundy, Provence, Aquitaine and a crowd of counties; they seek the Empire first
#                     6     the last war: 1400 came, Paris is French, England is still on the continent - 3 years of war
#                           (25 years at most) decide it
#                     7     stalemate: nobody won the last war - England loses its guarantees in France (control 0,
#                           revolts), France is worn out for 20 years, the two crowns leave each other alone
#                     1     French victory: England is off the continent - an island power, no colonies for 50 years;
#                           France tries a landing (15 years)
#                     2     French foothold: France holds two locations in Britain - 20 years for both sides, England
#                           strikes in the last 15; a war running at the end is waited out
#                     8     the break-up of England: the foothold held - England falls into its historical lands
#                           (Northumberland, Lancaster, Cornwall, Wales), they seek to reunite, Scotland's hour,
#                           English colonies only after 100 years
#   gp_sc_hyw_years   years in the current state (the clocks)
#   gp_sc_hyw_war_years   years of the last war fought (state 6)
#   gp_sc_hyw_landing_done   France has made or given up its landing (state 1)
#   gp_sc_old_ambitions   1 England took up the old kings' ambitions (25 %, once, 100 years after its defeat, if much
#                     stronger than France), 2 it did not
#   gp_sc_france_broken / gp_sc_england_broken   the crown has been divided (once a game; also by a crisis of 120)
# Country variables: gp_sc_island_power (England after state 1), gp_sc_rival_crown, gp_sc_truce_with (state 7),
# gp_sc_no_colonies_years (countdown), gp_sc_scot_hour_years (countdown).
# Modifiers obey the game rule "Scenario marks" (gp_rule_scenario_mods); the break-ups happen whatever the rules.
#
# v0.12.1 (observation 1337-1466, author 2026-11-10: "France landed in Wales; England a bit stronger, France slightly weaker"):
#   * the war goal of the vanilla Hundred Years' War no longer allows Welsh, Irish and northern English land (in_game/common/
#     wargoals/gp_wargoals.txt) - the war is about France;
#   * the machine calls gp_sc_prepare_hyw, the old unguarded form of gp_sc_prepare (gp_scenarios_west.txt): a pressed crown is
#     not forced into "preparation" by the other scenarios any more, but the posture forcing of the Hundred Years' War ignores
#     pressure (only that: the yearly posture pass and the 0.2 damping of the two forced lists still read it, see gp_sc_prepare);
#   * state 6 gives each crown a small, temporary, historically named nudge for 20 years (gp_sc_lancastrian_host,
#     gp_sc_armagnac_feud; AI crowns only, under the rules "Scenario marks" and "Nudges"; the author may veto each one), removed
#     again when the state ends (gp_sc_hyw_last_war_end);
#   * state 7 closes the vanilla situation with its own stalemate outcome (gp_sc_hyw_end_vanilla): the vanilla monthly lottery
#     (hundred_years_war.10 / .11 / .20 / .21) handed France a war on England in 1455, in the middle of "the crowns leave each
#     other alone". A CRUTCH (a scripted order, docs 3a); the evaluation route would be a weight on the vanilla event option,
#     which needs an override of the vanilla event.

# ---- the crowns ----------------------------------------------------------------------------------

# Anchor scope (the owner of London): saves scope:gp_eng and scope:gp_fra.
gp_sc_save_crowns = {
	if = {
		limit = { NOT = { has_global_variable = gp_sc_eng_crown } }
		location:london.owner ?= { save_scope_as = gp_crown_tmp }
		if = {
			limit = { exists = scope:gp_crown_tmp }
			set_global_variable = { name = gp_sc_eng_crown value = scope:gp_crown_tmp }
		}
		clear_saved_scope = gp_crown_tmp
	}
	if = {
		limit = { NOT = { has_global_variable = gp_sc_fra_crown } }
		location:paris.owner ?= { save_scope_as = gp_crown_tmp }
		if = {
			limit = { exists = scope:gp_crown_tmp }
			set_global_variable = { name = gp_sc_fra_crown value = scope:gp_crown_tmp }
		}
		clear_saved_scope = gp_crown_tmp
	}
	if = {
		limit = { global_var:gp_sc_eng_crown ?= { gp_is_real_country = yes } }
		global_var:gp_sc_eng_crown ?= { save_scope_as = gp_eng }
	}
	else = {
		location:london.owner ?= { save_scope_as = gp_eng }
	}
	if = {
		limit = { global_var:gp_sc_fra_crown ?= { gp_is_real_country = yes } }
		global_var:gp_sc_fra_crown ?= { save_scope_as = gp_fra }
	}
	else = {
		location:paris.owner ?= { save_scope_as = gp_fra }
	}
}

# ---- helpers ---------------------------------------------------------------------------------------

# Share of area $AREA$ held by scope:gp_eng or its subjects, into variable $VAR$ of the anchor.
gp_sc_area_share = {
	set_variable = { name = gp_sc_tmp_total value = 0 }
	set_variable = { name = gp_sc_tmp_owned value = 0 }
	area:$AREA$ = {
		every_location_in_area = {
			limit = { is_ownable = yes }
			scope:gp_c = { change_variable = { name = gp_sc_tmp_total add = 1 } }
			if = {
				limit = {
					owner ?= {
						OR = {
							this = scope:gp_eng
							is_subject_of = scope:gp_eng
						}
					}
				}
				scope:gp_c = { change_variable = { name = gp_sc_tmp_owned add = 1 } }
			}
		}
	}
	set_variable = {
		name = $VAR$
		value = {
			value = var:gp_sc_tmp_owned
			divide = {
				value = var:gp_sc_tmp_total
				min = 1
			}
		}
	}
	remove_variable = gp_sc_tmp_total
	remove_variable = gp_sc_tmp_owned
}

gp_sc_hyw_enter = {
	set_global_variable = { name = gp_sc_hyw value = $CODE$ }
	set_global_variable = { name = gp_sc_hyw_years value = 0 }
}

# Country scope: forget a prepared war against scope:$TARGET$.
gp_sc_drop_target = {
	if = {
		limit = { var:gp_sc_cb_target ?= scope:$TARGET$ }
		remove_variable = gp_sc_cb_target
	}
	if = {
		limit = { var:gp_war_target ?= scope:$TARGET$ }
		remove_variable = gp_war_target
	}
}

# scope:$WHO$ keeps preparing the war against scope:$TARGET$ while the state lasts.
gp_sc_hyw_keep_at_it = {
	scope:$WHO$ = {
		if = {
			limit = {
				NOT = { this = scope:$TARGET$ }
				NOT = { has_variable = gp_war_target }
			}
			gp_sc_prepare_hyw = { TARGET = $TARGET$ }
		}
	}
}

# Country scope: no colonies for $YEARS$ years (AI: the vanilla "avoid colonisation" preferences stay, gp_sc_overseas;
# everyone: colonial range -100 % under the rule "Scenario marks").
gp_sc_ban_colonies = {
	set_variable = { name = gp_sc_no_colonies_years value = $YEARS$ }
	if = {
		limit = { NOT = { has_game_rule = gp_scenario_mods_off } }
		if = {
			limit = { has_country_modifier = gp_sc_island_retreat }
			remove_country_modifier = gp_sc_island_retreat
		}
		add_country_modifier = { modifier = gp_sc_island_retreat years = $YEARS$ }
	}
}

# The state machine (anchor scope, scope:gp_eng and scope:gp_fra saved)
# ================================================================================================

gp_sc_hundred_years_war = {
	if = {
		limit = { has_global_variable = gp_sc_hyw_years }
		set_global_variable = {
			name = gp_sc_hyw_years
			value = {
				value = global_var:gp_sc_hyw_years
				add = 1
			}
		}
	}
	if = {
		limit = { NOT = { has_global_variable = gp_sc_hyw } }
		gp_sc_hyw_from_open = yes
	}
	else_if = {
		limit = { global_var:gp_sc_hyw = 3 }
		gp_sc_hyw_from_advantage = yes
	}
	else_if = {
		limit = { global_var:gp_sc_hyw = 5 }
		gp_sc_hyw_from_dual = yes
	}
	else_if = {
		limit = { global_var:gp_sc_hyw = 6 }
		gp_sc_hyw_from_last_war = yes
	}
	else_if = {
		limit = { global_var:gp_sc_hyw = 1 }
		gp_sc_hyw_from_victory = yes
	}
	else_if = {
		limit = { global_var:gp_sc_hyw = 2 }
		gp_sc_hyw_from_foothold = yes
	}
	# 4, 7 and 8 are final: their effects live on in the upkeep, the land marks and the AI scores
}

# The war for the crown.
gp_sc_hyw_from_open = {
	if = {
		limit = { gp_sc_eng_holds_paris = yes }
		gp_sc_hyw_dual_monarchy = yes
	}
	else_if = {
		limit = {
			current_date >= 1350.1.1
			gp_sc_eng_off_continent = yes
		}
		gp_sc_hyw_french_victory = yes
	}
	else_if = {
		limit = { current_date >= 1400.1.1 }
		gp_sc_hyw_last_war = yes
	}
	else = {
		gp_sc_area_share = { AREA = brittany_area VAR = gp_sc_adv_1 }
		gp_sc_area_share = { AREA = normandy_area VAR = gp_sc_adv_2 }
		gp_sc_area_share = { AREA = picardy_area VAR = gp_sc_adv_3 }
		if = {
			limit = {
				var:gp_sc_adv_1 >= 0.5
				var:gp_sc_adv_2 >= 0.5
				var:gp_sc_adv_3 >= 0.5
			}
			gp_sc_hyw_english_advantage = yes
		}
		remove_variable = gp_sc_adv_1
		remove_variable = gp_sc_adv_2
		remove_variable = gp_sc_adv_3
	}
}

# 3 English advantage: France has until 1400.
gp_sc_hyw_from_advantage = {
	if = {
		limit = { gp_sc_eng_holds_paris = yes }
		gp_sc_hyw_dual_monarchy = yes
	}
	else_if = {
		limit = { gp_sc_eng_off_continent = yes }
		gp_sc_hyw_french_victory = yes
	}
	else_if = {
		limit = { current_date >= 1400.1.1 }
		gp_sc_hyw_last_war = yes
	}
	else = {
		gp_sc_hyw_keep_at_it = { WHO = gp_fra TARGET = gp_eng }
	}
}

# 5 dual monarchy: 30 years to free Paris (a war still running at the end is waited out).
gp_sc_hyw_from_dual = {
	if = {
		limit = { gp_sc_eng_holds_paris = no }
		if = {
			limit = { gp_sc_eng_off_continent = yes }
			gp_sc_hyw_french_victory = yes
		}
		else_if = {
			limit = { current_date < 1400.1.1 }
			gp_sc_hyw_enter = { CODE = 3 }
		}
		else = {
			gp_sc_hyw_last_war = yes
		}
	}
	else_if = {
		limit = {
			OR = {
				gp_sc_fra_crown_fallen = yes
				AND = {
					global_var:gp_sc_hyw_years >= 30
					gp_sc_hyw_at_war = no
				}
			}
		}
		gp_sc_hyw_french_collapse = yes
	}
	else = {
		gp_sc_hyw_keep_at_it = { WHO = gp_fra TARGET = gp_eng }
	}
}

# 6 the last war: three years of war decide it; if nobody has won - stalemate.
gp_sc_hyw_from_last_war = {
	if = {
		limit = { gp_sc_eng_holds_paris = yes }
		gp_sc_hyw_dual_monarchy = yes
	}
	else_if = {
		limit = { gp_sc_eng_off_continent = yes }
		gp_sc_hyw_french_victory = yes
	}
	else = {
		if = {
			limit = { gp_sc_hyw_at_war = yes }
			set_global_variable = {
				name = gp_sc_hyw_war_years
				value = {
					value = global_var:gp_sc_hyw_war_years
					add = 1
				}
			}
		}
		if = {
			limit = {
				gp_sc_hyw_at_war = no
				OR = {
					global_var:gp_sc_hyw_war_years >= 3
					global_var:gp_sc_hyw_years >= 25
				}
			}
			gp_sc_hyw_stalemate = yes
		}
		else = {
			gp_sc_hyw_keep_at_it = { WHO = gp_eng TARGET = gp_fra }
			gp_sc_hyw_keep_at_it = { WHO = gp_fra TARGET = gp_eng }
		}
	}
}

# 1 French victory: the landing (15 years to make it) and, a century later, the old kings' ambitions.
gp_sc_hyw_from_victory = {
	if = {
		limit = {
			NOT = { has_global_variable = gp_sc_hyw_landing_done }
			gp_sc_fra_in_britain = yes
		}
		gp_sc_hyw_foothold = yes
	}
	else_if = {
		limit = {
			NOT = { has_global_variable = gp_sc_hyw_landing_done }
			global_var:gp_sc_hyw_years >= 15
		}
		set_global_variable = { name = gp_sc_hyw_landing_done value = 1 }
		scope:gp_fra = { gp_sc_drop_target = { TARGET = gp_eng } }
	}
	if = {
		limit = {
			NOT = { has_global_variable = gp_sc_old_ambitions }
			global_var:gp_sc_hyw_years >= 100
		}
		set_variable = {
			name = gp_sc_tmp_str
			value = {
				value = "scope:gp_fra.country_strength"
				multiply = 1.5
			}
		}
		if = {
			limit = {
				NOT = { scope:gp_fra = scope:gp_eng }
				scope:gp_eng = { country_strength > scope:gp_c.var:gp_sc_tmp_str }
			}
			random_list = {
				25 = {
					set_global_variable = { name = gp_sc_old_ambitions value = 1 }
					scope:gp_eng = { gp_sc_prepare_hyw = { TARGET = gp_fra } }
				}
				75 = {
					set_global_variable = { name = gp_sc_old_ambitions value = 2 }
				}
			}
		}
		else_if = {
			limit = { global_var:gp_sc_hyw_years >= 150 }
			set_global_variable = { name = gp_sc_old_ambitions value = 2 }
		}
		remove_variable = gp_sc_tmp_str
	}
}

# 2 French foothold: 20 years; England strikes in the last fifteen.
gp_sc_hyw_from_foothold = {
	if = {
		limit = {
			NOT = {
				scope:gp_fra = {
					any_owned_location = { region = region:great_britain_region }
				}
			}
		}
		# England has thrown the French into the sea: back to the island power
		gp_sc_hyw_enter = { CODE = 1 }
		set_global_variable = { name = gp_sc_hyw_landing_done value = 1 }
		scope:gp_fra = { gp_sc_drop_target = { TARGET = gp_eng } }
		scope:gp_eng = { gp_sc_drop_target = { TARGET = gp_fra } }
	}
	else_if = {
		limit = {
			global_var:gp_sc_hyw_years >= 20
			gp_sc_hyw_at_war = no
		}
		gp_sc_hyw_english_collapse = yes
	}
	else_if = {
		limit = { global_var:gp_sc_hyw_years >= 5 }
		scope:gp_eng = {
			if = {
				limit = {
					NOT = { this = scope:gp_fra }
					NOT = { has_variable = gp_war_target }
				}
				gp_sc_prepare_hyw = { TARGET = gp_fra }
				if = {
					limit = {
						is_ai = yes
						gp_is_active = yes
					}
					set_variable = { name = gp_posture_left value = 15 }
				}
			}
		}
	}
}

# ---- entering the states ---------------------------------------------------------------------------

# 3 English advantage: the spirit of Crecy (land morale +10 % for 20 years), both crowns arm.
gp_sc_hyw_english_advantage = {
	gp_sc_hyw_enter = { CODE = 3 }
	scope:gp_eng = {
		if = {
			limit = { NOT = { has_game_rule = gp_scenario_mods_off } }
			add_country_modifier = { modifier = gp_sc_crecy years = 20 }
		}
		gp_sc_prepare_hyw = { TARGET = gp_fra }
	}
	scope:gp_fra = {
		gp_crisis_shock = { AMOUNT = 20 }
		gp_sc_prepare_hyw = { TARGET = gp_eng }
	}
}

# 5 dual monarchy: the French crown (or, without it, the strongest free French country) fights for Paris; the princes
# lean on it; England makes client states instead of annexing (gp_sc_puppeteer).
gp_sc_hyw_dual_monarchy = {
	gp_sc_hyw_enter = { CODE = 5 }
	gp_sc_hyw_last_war_end = yes	# v0.12.1
	gp_sc_pick_france_leader = yes
	global_var:gp_sc_france_leader ?= {
		save_scope_as = gp_fl
		gp_sc_prepare_hyw = { TARGET = gp_eng }
	}
	gp_sc_france_patrons = yes
	clear_saved_scope = gp_fl
	scope:gp_eng = { set_variable = gp_sc_puppeteer }
}

# 6 the last war.
# v0.12.1 The nudges (author veto points, docs): the Lancastrian host for England (morale +5 %, manpower +5 %) and the quarrel of
# Armagnacs and Burgundians for France (levies -5 %, morale -3 %), 20 years, event-driven (this state is entered only when 1400
# came, Paris is French and England is still on the continent), AI crowns only, removed by gp_sc_hyw_last_war_end. Model:
# gp_sc_hyw_english_advantage below.
gp_sc_hyw_last_war = {
	gp_sc_hyw_enter = { CODE = 6 }
	set_global_variable = { name = gp_sc_hyw_war_years value = 0 }
	scope:gp_eng = {
		gp_sc_prepare_hyw = { TARGET = gp_fra }
		if = {
			limit = {
				is_ai = yes
				NOT = { has_game_rule = gp_scenario_mods_off }
				NOT = { has_game_rule = gp_nudges_off }
			}
			add_country_modifier = { modifier = gp_sc_lancastrian_host years = 20 }
		}
	}
	scope:gp_fra = {
		gp_sc_prepare_hyw = { TARGET = gp_eng }
		if = {
			limit = {
				is_ai = yes
				NOT = { has_game_rule = gp_scenario_mods_off }
				NOT = { has_game_rule = gp_nudges_off }
			}
			add_country_modifier = { modifier = gp_sc_armagnac_feud years = 20 }
		}
	}
}

# v0.12.1 The nudges of the last war end with it: called when the state is left for the stalemate, the dual monarchy or the French
# victory (anchor scope, scope:gp_eng and scope:gp_fra saved; harmless in any other state, there is nothing to remove then).
gp_sc_hyw_last_war_end = {
	scope:gp_eng = {
		if = {
			limit = { has_country_modifier = gp_sc_lancastrian_host }
			remove_country_modifier = gp_sc_lancastrian_host
		}
	}
	scope:gp_fra = {
		if = {
			limit = { has_country_modifier = gp_sc_armagnac_feud }
			remove_country_modifier = gp_sc_armagnac_feud
		}
	}
}

# v0.12.1 THE CRUTCH of the Hundred Years' War (docs 3a): the vanilla situation is closed with its own stalemate outcome when the
# mod's stalemate begins. Kept in its own small effect so that a parse problem here cannot disable the state machine. Model:
# end_situation / is_situation_active in vanilla scripted_effects/situation_effects.txt (council_of_trent_force_end_effect); the
# outcome "HYW_STALEMATE" is the name in vanilla common/situations/hundred_years_war.txt (its event .1537 costs both crowns some
# prestige and its on_ended removes the CBs cb_hundred_years_war / cb_hundred_years_war_fra and the mutual opinion). UNVERIFIED in
# game: that the effect is valid in this country scope and that a closed situation does not start again (the log of the
# observation shows no restart of any ended situation).
gp_sc_hyw_end_vanilla = {
	if = {
		limit = { is_situation_active = situation:hundred_years_war }
		end_situation = { situation = situation:hundred_years_war outcome = "HYW_STALEMATE" }
	}
}

# 7 stalemate: England has lost its guarantees in France (the land marks below); an AI England turns its French lands
# into vassal duchies (Normandy, Brittany, Aquitaine, Picardy, Anjou) whose people keep rising; France is worn out for
# 20 years; the crowns leave each other alone (gp_score_hyw_truce) - but do not make friends.
gp_sc_hyw_stalemate = {
	gp_sc_hyw_enter = { CODE = 7 }
	gp_sc_hyw_last_war_end = yes	# v0.12.1 the nudges of the last war end with it (they must not stack with gp_sc_french_weariness)
	scope:gp_eng = {
		gp_sc_drop_target = { TARGET = gp_fra }
		set_variable = { name = gp_sc_truce_with value = scope:gp_fra }
		if = {
			limit = { has_country_modifier = gp_sc_crecy }
			remove_country_modifier = gp_sc_crecy
		}
		gp_sc_course = { POSTURE = consolidation YEARS = 10 }
		if = {
			limit = {
				is_ai = yes
				gp_is_active = yes
			}
			gp_sc_eng_french_vassals = yes
		}
	}
	scope:gp_fra = {
		gp_sc_drop_target = { TARGET = gp_eng }
		set_variable = { name = gp_sc_truce_with value = scope:gp_eng }
		gp_sc_course = { POSTURE = recovery YEARS = 20 }
		if = {
			limit = { NOT = { has_game_rule = gp_scenario_mods_off } }
			add_country_modifier = { modifier = gp_sc_french_weariness years = 20 }
		}
	}
	gp_sc_hyw_end_vanilla = yes	# v0.12.1 the crowns leave each other alone - the vanilla lottery of wars is closed too
}

# 4 the break-up of France: the crown keeps only the area of its capital; the princes join the Empire, the emperor is
# their patron, the great duchies contend to unite France again; England makes client states.
gp_sc_hyw_french_collapse = {
	gp_sc_hyw_enter = { CODE = 4 }
	if = {
		limit = { gp_sc_fra_crown_fallen = no }
		scope:gp_fra = { gp_sc_france_breakup = yes }
	}
	gp_sc_france_princes_to_empire = yes
	gp_sc_pick_france_leader = yes
	scope:gp_eng = { set_variable = gp_sc_puppeteer }
}

# 1 French victory: England becomes an island power (no ambitions on the continent beyond client states in the Low
# Countries and Germany, no colonies for 50 years); France prepares a landing.
gp_sc_hyw_french_victory = {
	gp_sc_hyw_enter = { CODE = 1 }
	gp_sc_hyw_last_war_end = yes	# v0.12.1

# $TAG$ gets our part of area $AREA$: a dead state is raised from cores we give it; a living one gets the land and, if
# it was our vassal, its freedom (a vassal of somebody else gets nothing). The area of our capital stays ours.
gp_sc_release_state = {
	if = {
		limit = {
			NOT = { tag = $TAG$ }
			exists = capital
			NOT = { capital.area = area:$AREA$ }
			any_owned_location = { area = area:$AREA$ }
		}
		if = {
			limit = { NOT = { c:$TAG$ ?= { gp_is_real_country = yes } } }
			every_owned_location = {
				limit = { area = area:$AREA$ }
				add_core = c:$TAG$
			}
			create_country_from_cores_in_our_locations = c:$TAG$
		}
		else_if = {
			limit = {
				c:$TAG$ = {
					OR = {
						is_subject = no
						is_subject_of = scope:gp_brk
					}
				}
			}
			every_owned_location = {
				limit = { area = area:$AREA$ }
				change_location_owner = c:$TAG$
			}
			if = {
				limit = { c:$TAG$ = { is_subject_of = scope:gp_brk } }
				cancel_subject = c:$TAG$
			}
		}
	}
}
