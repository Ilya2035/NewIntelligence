###################################
# New Intelligence: Geopolicy - core effects (prefix gp_)
#
# One yearly pass per real country (gp_on_actions.txt): home-region share -> momentum of conquest -> regional threat
# -> intruders and common front (one loop) -> role -> (AI only) posture and stage zones. Nothing runs monthly.
# Postures change at once on a lost war (on_ending_war) and on a new ruler (on_new_ruler).
# NO country modifiers (author's rule 2026-11-04: no buffs, only behaviour). Roles and postures are plain variables read
# by the AI's own war-target scoring (ai_scripted_expansion_score), target lists, diplomatic objectives and zones.
# The country is saved as scope:gp_c at the start of every entry point and used instead of root: in the game-start loop
# (every_country inside on_game_started_after_lobby) root is NOT the country (first in-game test, 2026-11-04).
###################################

# ---- entry points --------------------------------------------------------------------------------

# global scope (on_game_started_after_lobby)
gp_on_game_start_effect = {
	debug_log = "GP_REPORT game start: setting up every real country"
	gp_history_relations = yes
	gp_history_contenders = yes
	every_country = {
		limit = { gp_is_real_country = yes }
		gp_setup_country = yes
	}
	gp_history_postures = yes
	gp_apply_terrain = yes
	debug_log = "GP_REPORT game start: setup finished"
}

# country scope (game start, newly formed country, released country)
gp_setup_country = {
	gp_assign_zone_profile = yes
	gp_yearly_update = yes
}

# country scope (yearly_country_pulse, game start loop, decision "Geopolitical overview")
gp_yearly_update = {
	save_scope_as = gp_c
	gp_compute_region_share = yes
	gp_update_momentum = yes
	gp_update_threat = yes
	gp_update_calm = yes
	gp_update_contender = yes
	gp_update_region_relations = yes
	gp_age_defeat = yes
	# v0.2 relations between nations and rulers (gp_relations_effects.txt)
	gp_form_young_royals = yes
	gp_update_ruler_affinity = yes
	gp_update_foreign_overlord = yes
	gp_ensure_colonial_profile = yes
	gp_update_region_claim = yes
	gp_update_control = yes
	if = {
		limit = { is_ai = no }
		gp_update_balance = yes	# v0.3 data of the "Balance of power" tab (gp_region_effects.txt)
	}
	if = {
		limit = { gp_is_active = yes }
		gp_update_role = yes
		if = {
			limit = { is_ai = yes }
			gp_update_posture = yes
			gp_update_stage_zones = yes
		}
		else = {
			gp_clear_postures = yes
		}
	}
	else = {
		gp_clear_roles = yes
		gp_clear_postures = yes
	}
	gp_debug_report = yes
	clear_saved_scope = gp_c
}

# country scope (on_ending_war, fired once per side): a lost war sends the AI into recovery at once.
gp_on_war_end_effect = {
	gp_war_memory = yes
	if = {
		limit = { scope:loser ?= root }
		set_variable = gp_recent_defeat
		set_variable = { name = gp_defeat_years value = 6 }
		if = {
			limit = {
				is_ai = yes
				gp_is_active = yes
				has_variable = gp_region_share
			}
			save_scope_as = gp_c
			gp_choose_posture = yes
			clear_saved_scope = gp_c
		}
	}
}

# country scope (on_new_ruler): a new ruler re-draws the course of the country ("the ruler changes the posture").
gp_on_new_ruler_effect = {
	save_scope_as = gp_c
	# v0.2: the new ruler brings their own likes and dislikes at once (gp_relations_effects.txt)
	gp_form_young_royals = yes
	gp_update_ruler_affinity = yes
	if = {
		limit = {
			is_ai = yes
			gp_is_active = yes
			has_variable = gp_region_share
		}
		gp_choose_posture = yes
	}
	clear_saved_scope = gp_c
}

# ---- home region share ---------------------------------------------------------------------------

# Owned locations of the home region are counted by walking OUR locations (small for most countries); the region's
# total of ownable locations is counted once and cached until the capital moves to another region.
gp_compute_region_share = {
	set_variable = { name = gp_owned_tmp value = 0 }
	if = {
		limit = { exists = capital }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = gp_region_cached }
					NOT = { capital.region = var:gp_region_cached }
				}
			}
			set_variable = { name = gp_region_total value = 0 }
			set_variable = { name = gp_region_cached value = capital.region }
			capital.region = {
				every_ownable_location_in_region = {
					scope:gp_c = { change_variable = { name = gp_region_total add = 1 } }
				}
			}
		}
		every_owned_location = {
			limit = { region = scope:gp_c.capital.region }
			scope:gp_c = { change_variable = { name = gp_owned_tmp add = 1 } }
		}
	}
	if = {
		limit = { NOT = { has_variable = gp_region_total } }
		set_variable = { name = gp_region_total value = 1 }
	}
	set_variable = { name = gp_region_share value = gp_region_share_value }
	set_variable = { name = gp_region_pct value = gp_region_pct_value }	# v0.3 whole percent for the "Balance of power" tab
	remove_variable = gp_owned_tmp
}

# ---- momentum of conquest ----------------------------------------------------------------------------

# var:gp_momentum = last year's momentum x 0.7 + net locations gained this year; gp_is_expanding while it exceeds
# 4 + 4 % of the country. Read by neighbours (threat, intruder, containment) - fast conquerors alarm their region.
gp_update_momentum = {
	if = {
		limit = { NOT = { has_variable = gp_loc_prev } }
		set_variable = { name = gp_loc_prev value = num_locations }
		set_variable = { name = gp_momentum value = 0 }
	}
	set_variable = { name = gp_momentum value = gp_momentum_next }
	set_variable = { name = gp_loc_prev value = num_locations }
	set_variable = { name = gp_momentum_excess_v value = gp_momentum_excess }
	if = {
		limit = { var:gp_momentum_excess_v >= 0 }
		set_variable = gp_is_expanding
	}
	else_if = {
		limit = { has_variable = gp_is_expanding }
		remove_variable = gp_is_expanding
	}
	remove_variable = gp_momentum_excess_v
}

# ---- regional threat -----------------------------------------------------------------------------

# Strongest qualifying power present in our home region (fallback: among our neighbours) -> var:gp_threat,
# var:gp_threat_share = its relative_strength against us (0..1; 0.6 = 1.5x, 0.75 = 3x). Opinion of it is refreshed
# (remove + add, so it never stacks).
gp_update_threat = {
	if = {
		limit = { has_variable = gp_threat }
		var:gp_threat ?= {
			scope:gp_c = {
				remove_opinion = { target = prev modifier = gp_opinion_regional_threat }
			}
		}
		remove_variable = gp_threat
	}
	if = {
		limit = { has_variable = gp_threat_share }
		remove_variable = gp_threat_share
	}
	if = {
		limit = { exists = capital }
		capital.region = {
			ordered_present_country = {
				limit = { gp_is_threat_candidate = yes }
				order_by = country_strength
				max = 1
				save_scope_as = gp_found_threat
			}
		}
		if = {
			limit = { NOT = { exists = scope:gp_found_threat } }
			ordered_neighbor_country = {
				limit = { gp_is_threat_candidate = yes }
				order_by = country_strength
				max = 1
				save_scope_as = gp_found_threat
			}
		}
		if = {
			limit = { exists = scope:gp_found_threat }
			set_variable = { name = gp_threat value = scope:gp_found_threat }
			set_variable = {
				name = gp_threat_share
				value = "scope:gp_found_threat.relative_strength(scope:gp_c)"
			}
			add_opinion = { target = scope:gp_found_threat modifier = gp_opinion_regional_threat }
		}
		clear_saved_scope = gp_found_threat
	}
}

# ---- intruders and common front (one loop over the countries present in the home region) ----------------

# Intruder (gp_triggers.txt): opinion -30, flag gp_has_intruder. Front partner (same faith, same threat or at war with
# it): opinion +20, flag gp_has_front_partner. Both refreshed yearly; the front dissolves by itself when the threat
# fades - "they help while it pays, then quarrel again".
gp_update_region_relations = {
	if = { limit = { has_variable = gp_has_intruder } remove_variable = gp_has_intruder }
	if = { limit = { has_variable = gp_has_front_partner } remove_variable = gp_has_front_partner }
	if = {
		limit = { exists = capital }
		capital.region = {
			every_present_country = {
				limit = {
					NOT = { this = scope:gp_c }
					gp_is_real_country = yes
				}
				if = {
					limit = { gp_is_intruder_of = { COUNTRY = scope:gp_c } }
					scope:gp_c = {
						remove_opinion = { target = prev modifier = gp_opinion_foreign_intruder }
						add_opinion = { target = prev modifier = gp_opinion_foreign_intruder }
						set_variable = gp_has_intruder
					}
				}
				if = {
					limit = { gp_is_front_partner_of = { COUNTRY = scope:gp_c } }
					scope:gp_c = {
						remove_opinion = { target = prev modifier = gp_opinion_common_enemy }
						add_opinion = { target = prev modifier = gp_opinion_common_enemy }
						set_variable = gp_has_front_partner
					}
				}
				# v0.3 contenders of one region watch each other (gp_region_effects.txt)
				if = {
					limit = { gp_is_regional_rival_of = { COUNTRY = scope:gp_c } }
					scope:gp_c = {
						remove_opinion = { target = prev modifier = gp_opinion_regional_rival }
						add_opinion = { target = prev modifier = gp_opinion_regional_rival }
					}
				}
				# v0.2 kin next door (gp_relations_effects.txt header)
				if = {
					limit = {
						is_neighbor_of = scope:gp_c
						gp_shares_culture_group_with = { COUNTRY = scope:gp_c }
						NOT = { gp_is_regional_rival_of = { COUNTRY = scope:gp_c } }
					}
					scope:gp_c = {
						remove_opinion = { target = prev modifier = gp_opinion_kin }
						add_opinion = { target = prev modifier = gp_opinion_kin }
					}
				}
				# v0.2 the shock that opens a time of troubles: a great power of our region (our threat, a regional
				# power, hegemon or empire) has just lost a war (gp_defeat_years is 6 in the year of the defeat).
				if = {
					limit = {
						has_variable = gp_defeat_years
						var:gp_defeat_years >= 5
						OR = {
							this ?= scope:gp_c.var:gp_threat
							has_variable = gp_role_regional
							has_variable = gp_role_hegemon
							has_variable = gp_role_empire
						}
					}
					scope:gp_c = {
						set_variable = gp_region_turmoil
						set_variable = { name = gp_turmoil_years value = 5 }
					}
				}
			}
		}
	}
}

# ---- lost wars ----------------------------------------------------------------------------------------

gp_age_defeat = {
	if = {
		limit = { has_variable = gp_defeat_years }
		change_variable = { name = gp_defeat_years add = -1 }
		if = {
			limit = { var:gp_defeat_years <= 0 }
			remove_variable = gp_defeat_years
			if = { limit = { has_variable = gp_recent_defeat } remove_variable = gp_recent_defeat }
		}
	}
}

# ---- stages --------------------------------------------------------------------------------------

# Thresholds are literals here (see gp_values.txt header): 0.25 = gp_share_regional, 0.55 = gp_share_hegemon.
# Aggressive expansion opens the far zones earlier (share 0.35).
gp_update_stage_zones = {
	if = {
		limit = { var:gp_region_share >= 0.25 }
		gp_zones_add_sphere = yes
	}
	else = {
		gp_zones_remove_sphere = yes
	}
	if = {
		limit = {
			OR = {
				var:gp_region_share >= 0.55
				AND = {
					has_variable = gp_posture_expansion
					gp_ruler_warlike = yes
					var:gp_region_share >= 0.35
				}
			}
		}
		gp_zones_add_far = yes
	}
	else = {
		gp_zones_remove_far = yes
	}
}

# ---- roles (plain variables, invisible to the player) ----------------------------------------------

gp_update_role = {
	gp_clear_roles = yes
	if = {
		limit = {
			has_variable = gp_threat
			var:gp_threat_share >= 0.75
			var:gp_threat ?= { is_neighbor_of = scope:gp_c }
		}
		set_variable = gp_role_besieged
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					var:gp_region_share >= 0.55
					num_locations >= gp_empire_min_locations
				}
				AND = {
					country_rank = country_rank:rank_empire
					num_locations >= gp_empire_rank_min_locations
				}
			}
		}
		set_variable = gp_role_empire
	}
	else_if = {
		limit = { var:gp_region_share >= 0.55 }
		set_variable = gp_role_hegemon
	}
	else_if = {
		limit = {
			OR = {
				var:gp_region_share >= 0.25
				AND = {
					country_rank_level >= 3
					num_locations >= 40
				}
			}
		}
		set_variable = gp_role_regional
	}
	else_if = {
		limit = {
			num_locations <= gp_status_quo_max_locations
			country_rank_level <= 1
			NOT = { gp_is_steppe = yes }
			NOT = { ai_personality = ai_personality:ai_aggressive }
			NOT = { ai_personality = ai_personality:ai_expansionist }
			NOT = { ai_personality = ai_personality:ai_opportunistic }
		}
		set_variable = gp_role_status_quo
	}
	else = {
		set_variable = gp_role_consolidator
	}
}

gp_clear_roles = {
	if = { limit = { has_variable = gp_role_consolidator } remove_variable = gp_role_consolidator }
	if = { limit = { has_variable = gp_role_regional } remove_variable = gp_role_regional }
	if = { limit = { has_variable = gp_role_hegemon } remove_variable = gp_role_hegemon }
	if = { limit = { has_variable = gp_role_empire } remove_variable = gp_role_empire }
	if = { limit = { has_variable = gp_role_besieged } remove_variable = gp_role_besieged }
	if = { limit = { has_variable = gp_role_status_quo } remove_variable = gp_role_status_quo }
}

# ---- postures (AI only; plain variables gp_posture_*, see docs/GEOPOLICY_DESIGN.md section 3a) ----------

# Yearly: count the posture down; re-choose when it runs out or when something forces a change (lost war, a
# stronger enemy at the border, isolation no longer possible, the prepared target gone).
gp_update_posture = {
	if = {
		limit = { has_variable = gp_posture_left }
		change_variable = { name = gp_posture_left add = -1 }
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = gp_posture_left }
				var:gp_posture_left <= 0
				AND = {
					has_variable = gp_recent_defeat
					NOT = { has_variable = gp_posture_recovery }
				}
				AND = {
					has_variable = gp_posture_isolation
					NOT = { gp_isolation_possible = yes }
				}
				AND = {
					has_variable = gp_threat
					var:gp_threat_share >= 0.67
					var:gp_threat ?= { is_neighbor_of = scope:gp_c }
					NOT = { has_variable = gp_posture_defense }
					NOT = { has_variable = gp_posture_common_front }
				}
				AND = {
					has_variable = gp_posture_preparation
					NOT = { var:gp_war_target ?= { gp_is_real_country = yes } }
				}
			}
		}
		gp_choose_posture = yes
	}
}

# Scores every posture (gp_values.txt: state interest + ruler x crown power + estates + inertia + one random swing),
# takes the highest (vanilla argmax pattern: global_effects.txt compare_and_get_highest_variable_from_4) and draws a
# random duration, so no country replays the same script every campaign.
gp_choose_posture = {
	random_list = {
		1 = { set_variable = { name = gp_noise value = 1 } }
		1 = { set_variable = { name = gp_noise value = 2 } }
		1 = { set_variable = { name = gp_noise value = 3 } }
		1 = { set_variable = { name = gp_noise value = 4 } }
		1 = { set_variable = { name = gp_noise value = 5 } }
		1 = { set_variable = { name = gp_noise value = 6 } }
		1 = { set_variable = { name = gp_noise value = 7 } }
		1 = { set_variable = { name = gp_noise value = 8 } }
		1 = { set_variable = { name = gp_noise value = 9 } }
	}
	set_variable = { name = gp_v1 value = gp_ps_expansion }
	set_variable = { name = gp_v2 value = gp_ps_preparation }
	set_variable = { name = gp_v3 value = gp_ps_opportunism }
	set_variable = { name = gp_v4 value = gp_ps_consolidation }
	set_variable = { name = gp_v5 value = gp_ps_defense }
	set_variable = { name = gp_v6 value = gp_ps_front }
	set_variable = { name = gp_v7 value = gp_ps_recovery }
	set_variable = { name = gp_v8 value = gp_ps_peace }
	set_variable = { name = gp_v9 value = gp_ps_isolation }
	set_variable = { name = gp_best value = var:gp_v1 }
	set_variable = { name = gp_best_code value = 1 }
	if = { limit = { var:gp_best < var:gp_v2 } set_variable = { name = gp_best value = var:gp_v2 } set_variable = { name = gp_best_code value = 2 } }
	if = { limit = { var:gp_best < var:gp_v3 } set_variable = { name = gp_best value = var:gp_v3 } set_variable = { name = gp_best_code value = 3 } }
	if = { limit = { var:gp_best < var:gp_v4 } set_variable = { name = gp_best value = var:gp_v4 } set_variable = { name = gp_best_code value = 4 } }
	if = { limit = { var:gp_best < var:gp_v5 } set_variable = { name = gp_best value = var:gp_v5 } set_variable = { name = gp_best_code value = 5 } }
	if = { limit = { var:gp_best < var:gp_v6 } set_variable = { name = gp_best value = var:gp_v6 } set_variable = { name = gp_best_code value = 6 } }
	if = { limit = { var:gp_best < var:gp_v7 } set_variable = { name = gp_best value = var:gp_v7 } set_variable = { name = gp_best_code value = 7 } }
	if = { limit = { var:gp_best < var:gp_v8 } set_variable = { name = gp_best value = var:gp_v8 } set_variable = { name = gp_best_code value = 8 } }
	if = { limit = { var:gp_best < var:gp_v9 } set_variable = { name = gp_best value = var:gp_v9 } set_variable = { name = gp_best_code value = 9 } }
	# Preparation needs a target in our zones; without one the country simply expands.
	if = {
		limit = { var:gp_best_code = 2 }
		gp_pick_war_target = yes
		if = {
			limit = { NOT = { has_variable = gp_war_target } }
			set_variable = { name = gp_best_code value = 1 }
		}
	}
	if = { limit = { var:gp_best_code = 1 } gp_set_posture = { POSTURE = expansion } }
	else_if = { limit = { var:gp_best_code = 2 } gp_set_posture = { POSTURE = preparation } }
	else_if = { limit = { var:gp_best_code = 3 } gp_set_posture = { POSTURE = opportunism } }
	else_if = { limit = { var:gp_best_code = 4 } gp_set_posture = { POSTURE = consolidation } }
	else_if = { limit = { var:gp_best_code = 5 } gp_set_posture = { POSTURE = defense } }
	else_if = { limit = { var:gp_best_code = 6 } gp_set_posture = { POSTURE = common_front } }
	else_if = { limit = { var:gp_best_code = 7 } gp_set_posture = { POSTURE = recovery } }
	else_if = { limit = { var:gp_best_code = 8 } gp_set_posture = { POSTURE = peace } }
	else = { gp_set_posture = { POSTURE = isolation } }
	if = {
		limit = {
			NOT = { has_variable = gp_posture_preparation }
			has_variable = gp_war_target
		}
		remove_variable = gp_war_target
	}
	# Duration: 3-7 years by lot; defense and common front are re-checked after 2, recovery lasts 5-8, isolation 8.
	random_list = {
		1 = { set_variable = { name = gp_posture_left value = 3 } }
		1 = { set_variable = { name = gp_posture_left value = 4 } }
		1 = { set_variable = { name = gp_posture_left value = 5 } }
		1 = { set_variable = { name = gp_posture_left value = 6 } }
		1 = { set_variable = { name = gp_posture_left value = 7 } }
	}
	if = {
		limit = {
			OR = {
				has_variable = gp_posture_defense
				has_variable = gp_posture_common_front
			}
		}
		set_variable = { name = gp_posture_left value = 2 }
	}
	else_if = {
		limit = { has_variable = gp_posture_recovery }
		change_variable = { name = gp_posture_left add = 2 }
	}
	else_if = {
		limit = { has_variable = gp_posture_isolation }
		set_variable = { name = gp_posture_left value = 8 }
	}
	remove_variable = gp_noise
	remove_variable = gp_v1
	remove_variable = gp_v2
	remove_variable = gp_v3
	remove_variable = gp_v4
	remove_variable = gp_v5
	remove_variable = gp_v6
	remove_variable = gp_v7
	remove_variable = gp_v8
	remove_variable = gp_v9
	remove_variable = gp_best
	remove_variable = gp_best_code
}

gp_set_posture = {
	gp_clear_postures = yes
	set_variable = gp_posture_$POSTURE$
	gp_apply_personality = yes	# v0.3: the mentality follows the course (gp_region_effects.txt)
}

gp_clear_postures = {
	if = { limit = { has_variable = gp_posture_expansion } remove_variable = gp_posture_expansion }
	if = { limit = { has_variable = gp_posture_preparation } remove_variable = gp_posture_preparation }
	if = { limit = { has_variable = gp_posture_opportunism } remove_variable = gp_posture_opportunism }
	if = { limit = { has_variable = gp_posture_consolidation } remove_variable = gp_posture_consolidation }
	if = { limit = { has_variable = gp_posture_defense } remove_variable = gp_posture_defense }
	if = { limit = { has_variable = gp_posture_common_front } remove_variable = gp_posture_common_front }
	if = { limit = { has_variable = gp_posture_recovery } remove_variable = gp_posture_recovery }
	if = { limit = { has_variable = gp_posture_peace } remove_variable = gp_posture_peace }
	if = { limit = { has_variable = gp_posture_isolation } remove_variable = gp_posture_isolation }
}

# The war a preparing country gets ready for: the richest neighbour that is clearly weaker (below 0.8x our strength),
# holds land in our home region (steppe hordes: any neighbour), and is not a friend, partner, subject or under truce.
gp_pick_war_target = {
	if = { limit = { has_variable = gp_war_target } remove_variable = gp_war_target }
	ordered_neighbor_country = {
		limit = {
			gp_is_real_country = yes
			NOT = { is_subject_or_below_of = scope:gp_c }
			NOT = { scope:gp_c = { is_subject_or_below_of = prev } }
			NOT = { in_union_with = scope:gp_c }
			NOT = { has_truce_with = scope:gp_c }
			NOT = { gp_is_front_partner_of = { COUNTRY = scope:gp_c } }
			NOT = {
				has_mutual_scripted_relation = {
					type = relation_type:alliance
					target = scope:gp_c
				}
			}
			relative_strength = {
				target = scope:gp_c
				value < 0.45
			}
			OR = {
				any_owned_location = { region = scope:gp_c.capital.region }
				scope:gp_c = { gp_is_steppe = yes }
			}
		}
		order_by = country_tax_base
		max = 1
		save_scope_as = gp_found_target
	}
	if = {
		limit = { exists = scope:gp_found_target }
		set_variable = { name = gp_war_target value = scope:gp_found_target }
	}
	clear_saved_scope = gp_found_target
}

# ---- natural frontiers (game rule; location modifiers for everyone, applied once at game start) -------

# Mountains: fort defense +100 % (author: "a fortress in the mountains gets 100 %"); hills +30 %; the open steppe is
# hard to hold: fort defense -25 %, garrisons refill slower. Static location modifiers in
# main_menu/common/static_modifiers/gp_terrain_modifiers.txt.
gp_apply_terrain = {
	if = {
		limit = { NOT = { has_game_rule = gp_terrain_off } }
		continent:europe = { every_location_in_continent = { gp_terrain_location = yes } }
		continent:asia = { every_location_in_continent = { gp_terrain_location = yes } }
		continent:africa = { every_location_in_continent = { gp_terrain_location = yes } }
		continent:america = { every_location_in_continent = { gp_terrain_location = yes } }
		continent:oceania = { every_location_in_continent = { gp_terrain_location = yes } }
	}
}

gp_terrain_location = {
	if = {
		limit = { topography = mountains }
		add_location_modifier = { modifier = gp_mountain_stronghold days = -1 }
	}
	else_if = {
		limit = { topography = hills }
		add_location_modifier = { modifier = gp_hill_stronghold days = -1 }
	}
	else_if = {
		limit = {
			topography = flatland
			OR = {
				region = region:steppes_region
				region = region:mongolia_region
				area = area:desht_kipchak_area
				area = area:zhetysu_area
			}
		}
		add_location_modifier = { modifier = gp_open_steppe days = -1 }
	}
}
