###################################
# New Intelligence: Geopolicy - core effects (prefix gp_)
#
# One yearly pass per real country (gp_on_actions.txt): home-region share -> regional threat -> intruders ->
# common front -> role (plain variables) -> (AI only) stage zones. Nothing runs monthly.
# NO country modifiers (author's rule 2026-11-04: no buffs, only behaviour). What a role changes is read by the AI's own
# war-target scoring (ai_scripted_expansion_score), target lists, diplomatic objectives and zones of interest.
# The country is saved as scope:gp_c at the start of gp_yearly_update and used instead of root: in the game-start loop
# (every_country inside on_game_started_after_lobby) root is NOT the country (first in-game test, 2026-11-04).
###################################

# ---- entry points --------------------------------------------------------------------------------

# global scope (on_game_started_after_lobby)
gp_on_game_start_effect = {
	debug_log = "GP_REPORT game start: setting up every real country"
	every_country = {
		limit = { gp_is_real_country = yes }
		gp_setup_country = yes
	}
	debug_log = "GP_REPORT game start: setup finished"
}

# country scope (game start, newly formed country, released country)
gp_setup_country = {
	gp_assign_zone_profile = yes
	gp_yearly_update = yes
}

# country scope (yearly_country_pulse, game start loop, decision "Geopolitical overview")
gp_yearly_update = {
	save_scope_as = gp_c
	gp_compute_region_share = yes
	gp_update_threat = yes
	gp_update_intruders = yes
	gp_update_common_front = yes
	if = {
		limit = { gp_is_active = yes }
		gp_update_role = yes
		if = {
			limit = { is_ai = yes }
			gp_update_stage_zones = yes
		}
	}
	else = {
		gp_clear_roles = yes
	}
	gp_debug_report = yes
	clear_saved_scope = gp_c
}

# ---- home region share ---------------------------------------------------------------------------

# Counts the ownable locations of the home region (effect-side counting: change_variable per location is the
# verified form; a list loop inside a script value with a scope change is not).
gp_compute_region_share = {
	set_variable = { name = gp_owned_tmp value = 0 }
	set_variable = { name = gp_total_tmp value = 0 }
	if = {
		limit = { exists = capital }
		capital.region = {
			every_ownable_location_in_region = {
				scope:gp_c = { change_variable = { name = gp_total_tmp add = 1 } }
				if = {
					limit = { owner ?= scope:gp_c }
					scope:gp_c = { change_variable = { name = gp_owned_tmp add = 1 } }
				}
			}
		}
	}
	set_variable = { name = gp_region_share value = gp_region_share_value }
	remove_variable = gp_owned_tmp
	remove_variable = gp_total_tmp
}

# ---- regional threat -----------------------------------------------------------------------------

# Strongest qualifying power present in our home region (fallback: among our neighbours) -> var:gp_threat,
# var:gp_threat_share = its relative_strength against us (0..1; 0.6 = 1.5x, 0.75 = 3x). Opinion of it is refreshed
# (remove + add, so it never stacks).
gp_update_threat = {
	if = {
		limit = { has_variable = gp_threat }
		var:gp_threat ?= {
			scope:gp_c = {
				remove_opinion = { target = prev modifier = gp_opinion_regional_threat }
			}
		}
		remove_variable = gp_threat
	}
	if = {
		limit = { has_variable = gp_threat_share }
		remove_variable = gp_threat_share
	}
	if = {
		limit = { exists = capital }
		capital.region = {
			ordered_present_country = {
				limit = { gp_is_threat_candidate = yes }
				order_by = country_strength
				max = 1
				save_scope_as = gp_found_threat
			}
		}
		if = {
			limit = { NOT = { exists = scope:gp_found_threat } }
			ordered_neighbor_country = {
				limit = { gp_is_threat_candidate = yes }
				order_by = country_strength
				max = 1
				save_scope_as = gp_found_threat
			}
		}
		if = {
			limit = { exists = scope:gp_found_threat }
			set_variable = { name = gp_threat value = scope:gp_found_threat }
			set_variable = {
				name = gp_threat_share
				value = "scope:gp_found_threat.relative_strength(scope:gp_c)"
			}
			add_opinion = { target = scope:gp_found_threat modifier = gp_opinion_regional_threat }
		}
		clear_saved_scope = gp_found_threat
	}
}

# ---- foreign intruders ---------------------------------------------------------------------------

# Every "intruder" (gp_triggers.txt) holding land in our home region: opinion -30, refreshed yearly.
gp_update_intruders = {
	if = {
		limit = { exists = capital }
		capital.region = {
			every_present_country = {
				limit = { gp_is_intruder_of = { COUNTRY = scope:gp_c } }
				scope:gp_c = {
					remove_opinion = { target = prev modifier = gp_opinion_foreign_intruder }
					add_opinion = { target = prev modifier = gp_opinion_foreign_intruder }
				}
			}
		}
	}
}

# ---- common front --------------------------------------------------------------------------------

# Countries of our region that name the same threat and share our religion group: +20 opinion (yearly refresh).
gp_update_common_front = {
	if = {
		limit = {
			exists = capital
			has_variable = gp_threat
		}
		capital.region = {
			every_present_country = {
				limit = {
					NOT = { this = scope:gp_c }
					gp_is_real_country = yes
					gp_shares_threat_with = { COUNTRY = scope:gp_c }
					religion.group = scope:gp_c.religion.group
					NOT = { this ?= scope:gp_c.var:gp_threat }
				}
				scope:gp_c = {
					remove_opinion = { target = prev modifier = gp_opinion_common_enemy }
					add_opinion = { target = prev modifier = gp_opinion_common_enemy }
				}
			}
		}
	}
}

# ---- stages --------------------------------------------------------------------------------------

# Thresholds are literals here (see gp_values.txt header): 0.25 = gp_share_regional, 0.55 = gp_share_hegemon.
gp_update_stage_zones = {
	if = {
		limit = { var:gp_region_share >= 0.25 }
		gp_zones_add_sphere = yes
	}
	else = {
		gp_zones_remove_sphere = yes
	}
	if = {
		limit = { var:gp_region_share >= 0.55 }
		gp_zones_add_far = yes
	}
	else = {
		gp_zones_remove_far = yes
	}
}

# ---- roles (plain variables, invisible to the player) ----------------------------------------------

gp_update_role = {
	gp_clear_roles = yes
	if = {
		limit = {
			has_variable = gp_threat
			var:gp_threat_share >= 0.75
			var:gp_threat ?= { is_neighbor_of = scope:gp_c }
		}
		set_variable = gp_role_besieged
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					var:gp_region_share >= 0.55
					num_locations >= gp_empire_min_locations
				}
				AND = {
					country_rank = country_rank:rank_empire
					num_locations >= gp_empire_rank_min_locations
				}
			}
		}
		set_variable = gp_role_empire
	}
	else_if = {
		limit = { var:gp_region_share >= 0.55 }
		set_variable = gp_role_hegemon
	}
	else_if = {
		limit = {
			OR = {
				var:gp_region_share >= 0.25
				AND = {
					country_rank_level >= 3
					num_locations >= 40
				}
			}
		}
		set_variable = gp_role_regional
	}
	else_if = {
		limit = {
			num_locations <= gp_status_quo_max_locations
			country_rank_level <= 1
			NOT = { gp_is_steppe = yes }
			NOT = { ai_personality = ai_personality:ai_aggressive }
			NOT = { ai_personality = ai_personality:ai_expansionist }
			NOT = { ai_personality = ai_personality:ai_opportunistic }
		}
		set_variable = gp_role_status_quo
	}
	else = {
		set_variable = gp_role_consolidator
	}
}

gp_clear_roles = {
	if = { limit = { has_variable = gp_role_consolidator } remove_variable = gp_role_consolidator }
	if = { limit = { has_variable = gp_role_regional } remove_variable = gp_role_regional }
	if = { limit = { has_variable = gp_role_hegemon } remove_variable = gp_role_hegemon }
	if = { limit = { has_variable = gp_role_empire } remove_variable = gp_role_empire }
	if = { limit = { has_variable = gp_role_besieged } remove_variable = gp_role_besieged }
	if = { limit = { has_variable = gp_role_status_quo } remove_variable = gp_role_status_quo }
}

# ---- diagnostics (test builds) --------------------------------------------------------------------

# A block of "GP_REPORT ..." lines in debug.log per marquee country (gp_is_debug_country) at game start and every year.
# First test (2026-11-04): debug_log loc cannot see saved scopes ([SCOPE.sCountry('x')] printed a null country), so the
# country is named by a literal tag (switch on tag, verified in game) and values are printed as literal buckets. One
# "loc check" line probes whether [ROOT...] works (yes in the yearly pulse expected, no at game start).
# GP TODO remove the call in gp_yearly_update before a public release.
gp_debug_report = {
	if = {
		limit = { gp_is_debug_country = yes }
		switch = {
			trigger = tag
			BRA = { debug_log = "GP_REPORT BRA Brandenburg" }
			HUN = { debug_log = "GP_REPORT HUN Hungary" }
			TUR = { debug_log = "GP_REPORT TUR Ottomans" }
			FRA = { debug_log = "GP_REPORT FRA France" }
			ENG = { debug_log = "GP_REPORT ENG England" }
			POL = { debug_log = "GP_REPORT POL Poland" }
			MOS = { debug_log = "GP_REPORT MOS Muscovy" }
			CAS = { debug_log = "GP_REPORT CAS Castile" }
			BYZ = { debug_log = "GP_REPORT BYZ Byzantium" }
			SER = { debug_log = "GP_REPORT SER Serbia" }
			BOH = { debug_log = "GP_REPORT BOH Bohemia" }
			HAB = { debug_log = "GP_REPORT HAB Austria" }
			LIT = { debug_log = "GP_REPORT LIT Lithuania" }
			DAN = { debug_log = "GP_REPORT DAN Denmark" }
			MAM = { debug_log = "GP_REPORT MAM Mamluks" }
			SCO = { debug_log = "GP_REPORT SCO Scotland" }
			NOV = { debug_log = "GP_REPORT NOV Novgorod" }
			VEN = { debug_log = "GP_REPORT VEN Venice" }
		}
		if = { limit = { var:gp_region_share >= 0.55 } debug_log = "GP_REPORT    region share >= 0.55" }
		else_if = { limit = { var:gp_region_share >= 0.25 } debug_log = "GP_REPORT    region share 0.25 - 0.55" }
		else_if = { limit = { var:gp_region_share >= 0.1 } debug_log = "GP_REPORT    region share 0.10 - 0.25" }
		else = { debug_log = "GP_REPORT    region share < 0.10" }
		if = {
			limit = { has_variable = gp_threat }
			if = { limit = { var:gp_threat_share >= 0.75 } debug_log = "GP_REPORT    threat found, 3x or more" }
			else_if = { limit = { var:gp_threat_share >= 0.67 } debug_log = "GP_REPORT    threat found, 2x - 3x" }
			else = { debug_log = "GP_REPORT    threat found, 1.5x - 2x" }
		}
		else = { debug_log = "GP_REPORT    no regional threat" }
		if = { limit = { has_variable = gp_has_profile } debug_log = "GP_REPORT    zone profile historical" }
		else = { debug_log = "GP_REPORT    zone profile generic" }
		if = { limit = { has_variable = gp_role_consolidator } debug_log = "GP_REPORT    role consolidator" }
		if = { limit = { has_variable = gp_role_regional } debug_log = "GP_REPORT    role regional power" }
		if = { limit = { has_variable = gp_role_hegemon } debug_log = "GP_REPORT    role hegemon" }
		if = { limit = { has_variable = gp_role_empire } debug_log = "GP_REPORT    role empire" }
		if = { limit = { has_variable = gp_role_besieged } debug_log = "GP_REPORT    role besieged" }
		if = { limit = { has_variable = gp_role_status_quo } debug_log = "GP_REPORT    role status quo" }
		if = { limit = { is_ai = no } debug_log = "GP_REPORT    player country" }
		debug_log = "GP_REPORT    loc check: root [ROOT.GetCountry.GetName], threat [ROOT.GetCountry.MakeScope.GetVariable('gp_threat').GetCountry.GetName], share [ROOT.GetCountry.MakeScope.GetVariable('gp_region_share').GetValue|2]"
	}
}
