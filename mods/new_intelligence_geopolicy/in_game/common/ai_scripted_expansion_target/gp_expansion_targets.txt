# New Intelligence: Geopolicy - scripted expansion targets (prefix gp_)
#
# Format: ai_scripted_expansion_target/readme.txt. attacker_potential and candidate_list: root = attacker;
# score / sort_value: scope:attacker, scope:target. A target is picked when its score reaches
# EXPANSION_TARGET_SCORE_NEEDED_TO_PICK (25 with gp_defines.txt); vanilla attack_neighbors scores tax base / distance * 10.
# Strength values are relative_strength shares 0..1.
#
# v0.12.1 (diag D4, D6) PRESSURE DAMPING. The lists that history or a scenario prepares - the prepared war, the intruder,
# the unification, the scenario war and vassal, the dynastic claim and the two Eastern lists at the end - carry a fixed score
# of 20-28 + tax/100, and 25 alone is the pick threshold: a beaten or threatened country was carried to a war by that score
# alone (Castile attacked Aragon and Portugal while it was losing). Each of them now loses 0.2 per point of the attacker's
# PRESSURE (scope:attacker.gp_pressure_load, 0..100, script_values/gp_values.txt: lost wars, a threat on its soil, exhaustion;
# pressed from 40, hot from 45): from ~8 points on a prepared war no longer reaches the threshold by itself; the engine's own desire can still
# lift it. Not damped: contain_threat (the pressed defend), home_region (tax/distance, no fixed score), horde_tribute,
# colonial_zone, and outer_dominion (it has its own rule: -12 for a power with under a quarter of its home region). The factor
# 0.2 is the same in every damped list - one number to retune (search "gp_pressure_load"). Beside it gp_score_pressure_restraint
# (gp_expansion_scores.txt) takes score and multiplier off every target that is not the regional threat.
# v0.12.1 the lists "throne claim" and "kin neighbour" are new (diag D4): the Eastern road, see the end of the file.
# Next observation: no war declared by a country with GP_REPORT "pressure >= 40" on a target that is not its threat (Castile
# 1413-65); no Polish-Lithuanian war (1365); no Spremberg / Aquileia / Holstein client wars; Poland -> Halych / Volhynia.
###################################

# 1. The war a preparing country got ready for (gp_posture_preparation, var:gp_war_target chosen in gp_pick_war_target).
gp_target_prepared_war = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_posture_preparation
		has_variable = gp_war_target
		at_war = no
		# v0.12.1 (D6) no new war on the heels of the last: Castile attacked Portugal in 1395 right after a lost war. A
		# CRUTCH (flagged): the cooldown is a gate; the replacement is the personality's own ai_months_between_wars plus the
		# exhaustion terms of the pressure model.
		months_since_war >= 18
	}
	candidate_list = {
		var:gp_war_target ?= {
			if = {
				limit = {
					gp_is_real_country = yes
					NOT = { has_truce_with = root }
				}
				add_to_list = source
			}
		}
	}
	score = {
		add = 26
		add = {
			value = scope:target.country_tax_base
			divide = 100
		}
		subtract = {	# v0.12.1 pressure damping (see the header)
			value = scope:attacker.gp_pressure_load
			multiply = 0.2
		}
	}
	sort_value = {
		add = 900
	}
}

# 2. Contain the regional threat (mod casus belli, cuts the target down to size) - only while it presses on the region.
gp_target_contain_threat = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_threat
		var:gp_threat_share <= 0.71		# at most 2.5x as strong (relative_strength share)
		gp_threat_is_pressing = yes
		months_since_war >= 24
		NOT = { has_variable = gp_role_status_quo }
		NOT = { gp_is_calm_posture = yes }
	}
	candidate_list = {
		var:gp_threat ?= {
			if = {
				limit = {
					gp_is_real_country = yes
					NOT = { has_truce_with = root }
				}
				add_to_list = source
			}
		}
	}
	casus_belli = casus_belli:gp_cb_contain_threat
	score = {
		add = 25
		add = {
			value = scope:target.country_tax_base
			divide = 100
		}
	}
	sort_value = {
		add = 1000
	}
}

# 3. Expel the intruder: foreign or conquering powers holding land in our home region - only those weaker than we
#    are, and only in an active posture (second test: at game start almost every European country has an intruder,
#    an open list would drive the AI back into constant wars).
gp_target_expel_intruder = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
		months_since_war >= 24
		NOT = { has_variable = gp_role_status_quo }
		NOT = { has_variable = gp_role_besieged }
		OR = {
			has_variable = gp_posture_expansion
			has_variable = gp_posture_opportunism
		}
	}
	candidate_list = {
		capital.region = {
			every_present_country = {
				limit = {
					gp_is_intruder = yes
					NOT = { has_truce_with = root }
					relative_strength = {
						target = root
						value < 0.5		# weaker than we are (share)
					}
				}
				add_to_list = source
			}
		}
	}
	score = {
		add = 20
		subtract = {	# v0.12.1 pressure damping (see the header)
			value = scope:attacker.gp_pressure_load
			multiply = 0.2
		}
	}
	sort_value = {
		add = 500
	}
}

# 4. The home region beyond our borders: weaker powers present in our region that are not direct neighbours
#    (across a strait, behind a vassal) - vanilla attack_neighbors only sees neighbours.
gp_target_home_region = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
		months_since_war >= 36
		gp_is_expansion_role = yes
		OR = {
			has_variable = gp_posture_expansion
			has_variable = gp_posture_opportunism
		}
	}
	candidate_list = {
		capital.region = {
			every_present_country = {
				limit = {
					NOT = { this = root }
					gp_is_real_country = yes
					NOT = { is_neighbor_of = root }
					NOT = { is_subject_or_below_of = root }
					NOT = { root = { is_subject_or_below_of = prev } }
					NOT = { in_union_with = root }
					NOT = { has_truce_with = root }
					relative_strength = {
						target = root
						value < 0.41		# weaker than 0.7x our strength (share)
					}
				}
				add_to_list = source
			}
		}
	}
	score = {
		add = scope:target.country_tax_base
		divide = "scope:target.capital.distance_to(scope:attacker.capital)"
		multiply = 10
	}
}

# 5. The horde collects tribute: settled neighbours outside the steppe become tributaries (vanilla cb_make_tributary,
#    usable by steppe hordes), steppe land itself is conquered through the zones of interest. Small bites, easy to hold.

	candidate_list = {
		every_neighbor_country = {
			limit = {
				gp_is_real_country = yes
				is_subject = no
				exists = capital
				NOT = { gp_is_steppe = yes }

				relative_strength = {
					target = root

				}
			}
			add_to_list = source
		}
	}
	casus_belli = casus_belli:cb_make_tributary
	score = {
		add = 24
	}
	sort_value = {
		add = 300
		add = scope:target.country_tax_base
	}
}

# 6. v0.2 Outer dominion: a power that holds its home region makes the weak neighbours beyond it its vassals instead of
#    annexing them (gp_cb_establish_dominion). Same faith group only - the infidel next door is conquered as before.
gp_target_outer_dominion = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
		# v0.4 (author: "Britain and France like to create subject states abroad and next to their borders, Moscow and
		# Poland would rather centralise"): Eastern Europe annexes, it does not make vassals beyond its region.
		NOT = { capital.sub_continent = sub_continent:eastern_europe }
		at_war = no
		OR = {
			# v0.7 Hundred Years' War branches 2, 4, 5: the victor makes puppets instead of annexing
			AND = {
				has_variable = gp_sc_puppeteer
				months_since_war >= 12
				NOT = { gp_is_calm_posture = yes }
			}
			AND = {
				gp_is_expansion_role = yes
				months_since_war >= 36
				OR = {
					has_variable = gp_posture_expansion
					has_variable = gp_posture_opportunism
				}
			}
		}
	}
	candidate_list = {
		every_neighbor_country = {
			limit = {
				gp_is_real_country = yes
				is_subject = no
				exists = capital
				NOT = { capital.region = root.capital.region }
				religion.group = root.religion.group
				NOT = { has_truce_with = root }
				NOT = { is_allied_with = { target = root } }
				NOT = { this ?= root.var:gp_ruler_like }
				NOT = { gp_is_old_friend_of = { COUNTRY = root } }
				relative_strength = {
					target = root
					value < 0.3		# weaker than ~0.43x the attacker
				}
			}
			add_to_list = source
		}
	}
	casus_belli = casus_belli:gp_cb_establish_dominion
	score = {
		add = 22
		if = {
			limit = { scope:attacker = { capital.sub_continent = sub_continent:western_europe } }
			add = 6	# the West builds a ring of client states
		}
		if = {
			limit = { scope:attacker = { has_variable = gp_sc_puppeteer } }
			add = 15	# v0.7 the puppeteer of a scenario
		}
		# v0.12.1 (diag D4) client states beyond the region are for a power that already holds a quarter of its own region (the
		# stage at which the sphere preferences open, gp_share_regional). The "ring of client states" bonus above went to every
		# Western power in an expansion role: Bohemia (10 % of South Germany) made the one-location lordship Spremberg its
		# client in 1358, Austria (17 %) Aquileia twice, Denmark three Holstein counties. A score malus, not a ban: 22 + 6 - 12
		# = 16 + tax/100 stays under the pick threshold 25, but a scenario (puppeteer +15) or the engine's own desire can still
		# lift it. One number, no list of countries: the active are those that hold their home region (France: >= 25 % all game,
		# England: >= 55 %; they keep their ring of client states).
		if = {
			limit = {
				scope:attacker = {
					has_variable = gp_region_share
					var:gp_region_share < 0.25
				}
			}
			add = -12
		}
		add = {
			value = scope:target.country_tax_base
			divide = 100
		}
	}
	sort_value = {
		add = 200
		add = scope:target.country_tax_base
	}
}

# 7. v0.3 Unite the region (gp_cb_regional_unification): a contender in a region in tension (or the 90 % holder with its
#    claim) goes for the weaker countries of its own home region - the internal wars that unite a region.
gp_target_regional_unification = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
		at_war = no
		months_since_war >= 24
		gp_has_unification_right = yes
		NOT = { gp_is_calm_posture = yes }
	}
	candidate_list = {
		every_neighbor_country = {
			limit = {
				gp_is_real_country = yes
				is_subject = no
				exists = capital
				capital.region = root.capital.region
				NOT = { has_truce_with = root }
				NOT = { is_allied_with = { target = root } }
				NOT = { this ?= root.var:gp_ruler_like }
				NOT = { this ?= root.var:gp_patron }
				# v0.12.1 (diag D4) old friends unite by crown, not by war: the only war Poland declared in 129 years was this
				# list's war on Lithuania (1365), the ally of 1325 (Aldona's marriage). A CRUTCH (flagged): the replacement is the
				# v0.15 "secret court" union (influence -> marriage -> union) of docs/GEOPOLICY_V012_PLAN.md; the same exclusion
				# already stands in gp_target_outer_dominion above.
				NOT = { gp_is_old_friend_of = { COUNTRY = root } }
				relative_strength = {
					target = root
					value < 0.45		# weaker than ~0.8x the attacker
				}
			}
			add_to_list = source
		}
	}
	casus_belli = casus_belli:gp_cb_regional_unification
	score = {
		add = 24
		add = {
			value = scope:target.country_tax_base
			divide = 100
		}
		subtract = {	# v0.12.1 the pressed unite nothing (pressure damping, see the header)
			value = scope:attacker.gp_pressure_load
			multiply = 0.2
		}
	}
	sort_value = {
		add = 500
		add = scope:target.country_tax_base
	}
}
