###################################
# New Intelligence: Geopolicy - scripted expansion scores (prefix gp_)
#
# Format: ai_scripted_expansion_score/readme.txt. attacker_potential: root = attacker; target_trigger: root = target,
# scope:attacker = attacker; score / multiplier: scope:attacker, scope:target.
# This file carries ALL of the mod's influence on whom the AI wants to fight: there are no country modifiers
# (author's rule 2026-11-04). Roles (var:gp_role_*) and temperaments (government / capital / faith) act here only.
# Convention (as vanilla gag_guelphs: multiplier = { add = -0.75 } "much less likely"): the multiplier is read as
# (1 + value) times the final score, so -0.4 = x0.6 and +0.3 = x1.3. GP TODO verify in the AI war-planning tooltip.
# A target is picked only when its final score reaches EXPANSION_TARGET_SCORE_NEEDED_TO_PICK = 20, so lowering the
# multiplier of every target is how a calm role "waits between wars" without any modifier.
# Strength values are relative_strength shares 0..1 (0.71 = 2.5x). Nothing uses never_attack.
###################################

# ================================================================================================
# Zones and neighbourhood
# ================================================================================================

# Zone discipline: a target that is neither a neighbour nor in our home region is worth less the farther it lies.
# Empires and steppe hordes are exempt (they think in sub-continents).
gp_score_distance_discipline = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
		NOT = { has_variable = gp_role_empire }
		NOT = { gp_is_steppe = yes }
	}
	target_trigger = {
		exists = capital
		NOT = { is_neighbor_of = scope:attacker }
		NOT = { capital.region = scope:attacker.capital.region }
	}
	multiplier = {
		add = 0
		if = {
			limit = { "scope:attacker.border_distance_to(scope:target)" > 250 }
			add = -0.7
		}
		else_if = {
			limit = { "scope:attacker.border_distance_to(scope:target)" > 100 }
			add = -0.4
		}
	}
}

# ================================================================================================
# Regional threat, common front, intruders
# ================================================================================================

# Common front: the countries that name the same regional threat as we do (and share our faith group) are not
# targets while that threat stands - "we are enemies, but for now it drives us together".
gp_score_common_front = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_threat
	}
	target_trigger = {
		gp_shares_threat_with = { COUNTRY = scope:attacker }
		religion.group = scope:attacker.religion.group
	}
	multiplier = {
		add = -0.6
	}
}

# Containment: our regional threat is a preferred target as long as it is at most 2.5x as strong (share 0.71);
# the engine's win-chance check still applies.
gp_score_contain_threat = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_threat
		var:gp_threat_share <= 0.71
	}
	target_trigger = {
		this ?= scope:attacker.var:gp_threat
	}
	score = {
		add = 15
	}
	multiplier = {
		add = 0.3
	}
}

# Expel the intruder: a power from another sub-continent / faith holding land in our home region.
gp_score_expel_intruder = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
	}
	target_trigger = {
		gp_is_intruder_of = { COUNTRY = scope:attacker }
	}
	score = {
		add = 10
	}
	multiplier = {
		add = 0.25
	}
}

# Holy restraint: while our regional threat is of another faith group, co-religionists are attacked less eagerly.
gp_score_heathen_threat_restraint = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_threat
		NOT = {
			var:gp_threat ?= { religion.group = root.religion.group }
		}
	}
	target_trigger = {
		religion.group = scope:attacker.religion.group
		NOT = { this ?= scope:attacker.var:gp_threat }
	}
	multiplier = {
		add = -0.35
	}
}

# ================================================================================================
# Roles (var:gp_role_*, set yearly by gp_update_role)
# ================================================================================================

# Consolidator: wants the land of its own region and its neighbours - more eager there.
gp_score_role_consolidator = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_role_consolidator
		exists = capital
	}
	target_trigger = {
		OR = {
			is_neighbor_of = scope:attacker
			any_owned_location = { region = scope:attacker.capital.region }
		}
	}
	multiplier = {
		add = 0.25
		if = {
			limit = { gp_is_strong_mode = yes }
			add = 0.15
		}
	}
}

# Hegemon: keeps the peace at home - after a war it waits (3 years) before the next one, and it prefers to dominate
# the small co-religionist neighbours of its region (subjects, influence) rather than to eat them.
gp_score_role_hegemon_pause = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_role_hegemon
		months_since_war < 36
	}
	target_trigger = {
		NOT = { this ?= scope:attacker.var:gp_threat }
	}
	multiplier = {
		add = -0.5
	}
}

gp_score_hegemon_restraint = {

		is_ai = yes
		gp_is_active = yes
		has_variable = gp_role_hegemon

	}
	target_trigger = {

	}

# Chinese, Japanese and Korean states value targets outside their home region very little.
gp_score_isolationist_civ = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
		OR = {
			gp_is_sinic = yes
			gp_is_japanese = yes
			gp_is_korean = yes
		}
	}
	target_trigger = {
		exists = capital

		NOT = { is_neighbor_of = scope:attacker }
	}
	multiplier = {
		add = -0.8
	}
}
