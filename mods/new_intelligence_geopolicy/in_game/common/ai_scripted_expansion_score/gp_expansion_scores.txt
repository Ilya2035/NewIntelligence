###################################
# New Intelligence: Geopolicy - scripted expansion scores (prefix gp_)
#
# Format: ai_scripted_expansion_score/readme.txt. attacker_potential: root = attacker; target_trigger: root = target,
# scope:attacker = attacker; score / multiplier: scope:attacker, scope:target.
# This file carries ALL of the mod's influence on whom the AI wants to fight: there are no country modifiers
# (author's rule 2026-11-04). Postures (var:gp_posture_*), roles (var:gp_role_*) and temperaments act here only.
# Convention (as vanilla gag_guelphs: multiplier = { add = -0.75 } "much less likely"): the multiplier is read as
# (1 + value) times the final score, so -0.4 = x0.6 and +0.3 = x1.3. GP TODO verify in the AI war-planning tooltip.
# A target is picked only when its final score reaches EXPANSION_TARGET_SCORE_NEEDED_TO_PICK (raised 20 -> 25 in
# gp_defines.txt), so lowering the multiplier of every target is how a calm posture "waits" without any modifier.
# Strength values are relative_strength shares 0..1 (0.71 = 2.5x). Nothing uses never_attack.
###################################

# Expansion: a little more eager for targets inside the home region and next door.
gp_score_posture_expansion = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		has_variable = gp_posture_expansion
		exists = capital
	}
	target_trigger = {
		OR = {
			is_neighbor_of = scope:attacker
			any_owned_location = { region = scope:attacker.capital.region }
		}
	}
	multiplier = {
		add = 0.15
	}
}

# Easy prey: a neighbour we dwarf (natives, petty lordships) is fair game in any non-calm posture - it costs little
# and alarms nobody much (author: "when it feels total superiority and no general outrage").
gp_score_light_expansion = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		NOT = { gp_is_calm_posture = yes }
	}
	target_trigger = {
		is_neighbor_of = scope:attacker
		gp_is_tiny_target_of_attacker = yes
	}
	score = {
		add = 5
	}
	multiplier = {
		add = 0.3
	}
}

# v0.3 The region first (author, 2026-11-05: "until 1600 countries work fully for their own region, and a little on the
# lands next to it"): a target with no land in our home region is worth a third as much (the old 20-year early restraint
# is folded in here). Who holds 75 % of the home region has earned the right to the neighbouring regions of its
# macro-region (x0.8 there). Hegemons, empires and steppe hordes are exempt; prepared wars and easy prey are not held back.
gp_score_region_first = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		gp_is_before_1600 = yes
		exists = capital
		NOT = { has_variable = gp_role_hegemon }
		NOT = { has_variable = gp_role_empire }
		NOT = { gp_is_steppe = yes }
	}
	target_trigger = {
		NOT = { any_owned_location = { region = scope:attacker.capital.region } }
		NOT = { gp_is_tiny_target_of_attacker = yes }
		NOT = { this ?= scope:attacker.var:gp_war_target }	# the historical openings (gp_history_postures) are not held back
		NOT = { this ?= scope:attacker.var:gp_sc_cb_target }	# v0.5 scenario wars are not held back
		NOT = {
			AND = {
				scope:attacker = { has_variable = gp_sc_ireland_right }	# v0.5 branch 2D: the quarrelling Gaels
				exists = capital
				capital.region = region:ireland_region
			}
		}
	}
	multiplier = {
		add = -0.65
		if = {
			limit = {
				scope:attacker = { var:gp_region_share >= 0.75 }
				scope:target = {
					exists = capital
					capital.sub_continent = scope:attacker.capital.sub_continent
				}
			}
			add = 0.45
		}
	}
}

# Zone discipline: a target that is neither a neighbour nor in our home region is worth less the farther it lies.
# Empires and steppe hordes are exempt (they think in sub-continents).
gp_score_distance_discipline = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		exists = capital
		NOT = { has_variable = gp_role_empire }
		NOT = { gp_is_steppe = yes }
	}
	target_trigger = {
		exists = capital
		NOT = { is_neighbor_of = scope:attacker }
		NOT = { capital.region = scope:attacker.capital.region }
	}
	multiplier = {
		add = 0
		if = {
			limit = { "scope:attacker.border_distance_to(scope:target)" > 250 }
			add = -0.7
		}
		else_if = {
			limit = { "scope:attacker.border_distance_to(scope:target)" > 100 }
			add = -0.4
		}
	}
}

# ================================================================================================
# v0.2 Status quo between peers (author: "two strong powers at peace for a long time keep the status quo, unless one
# of them is weakened")
# ================================================================================================

# Two powers of standing of comparable strength (0.67x..1.5x, shares 0.4..0.6) with no fresh war between them (the
# memory of a war, gp_opinion_old_enemy, fades in ~15 years per war) hold each other in check. A defeat, a regency, a
# war elsewhere or a time of troubles in the region breaks it.
gp_score_status_quo_peers = {
	attacker_potential = {
		is_ai = yes
		gp_is_active = yes
		at_war = no
		gp_is_power_of_standing = yes
		NOT = { has_variable = gp_region_turmoil }
	}
	target_trigger = {
		gp_is_power_of_standing = yes
		relative_strength = {
			target = scope:attacker
			value >= 0.4
		}
		relative_strength = {
			target = scope:attacker
			value <= 0.6
		}
		at_war = no
		has_regent = no
		NOT = { has_variable = gp_recent_defeat }
		NOT = { gp_is_old_enemy_of = { COUNTRY = scope:attacker } }
		NOT = { this ?= scope:attacker.var:gp_ruler_dislike }
		NOT = { this ?= scope:attacker.var:gp_war_target }
	}
	multiplier = {
		add = -0.5
	}
}

		relative_strength = {

			value < 0.35
		}
