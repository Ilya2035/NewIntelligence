gp_bp_tension = {
	type = country
	log_loc_errors = yes

	text = {
		localization_key = gp_bp_tension_turmoil
		trigger = { has_variable = gp_region_turmoil }
	}
	text = {
		localization_key = gp_bp_tension_tense
		trigger = {
			has_variable = gp_calm_years
			var:gp_calm_years >= 10
		}
	}
	text = {
		localization_key = gp_bp_tension_war
		trigger = { at_war = yes }
	}
	text = {
		localization_key = gp_bp_tension_calm
		fallback = yes
	}
}

gp_bp_cohesion = {
	type = country
	log_loc_errors = yes

	text = {
		localization_key = gp_bp_cohesion_united
		trigger = {
			var:gp_bp_holder ?= {
				has_variable = gp_region_share
				var:gp_region_share >= 0.9
			}
		}
	}
	text = {
		localization_key = gp_bp_cohesion_dominated
		trigger = {
			var:gp_bp_holder ?= {
				has_variable = gp_region_share
				var:gp_region_share >= 0.55
			}
		}
	}
	text = {
		localization_key = gp_bp_cohesion_contested
		trigger = {
			var:gp_bp_holder ?= {
				has_variable = gp_region_share
				var:gp_region_share >= 0.25
			}
		}
	}
	text = {
		localization_key = gp_bp_cohesion_fragmented
		fallback = yes
	}
}
