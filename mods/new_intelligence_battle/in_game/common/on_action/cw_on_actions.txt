# ---- our on_actions ------------------------------------------------------------------------

# Monthly bookkeeping. The cheap trigger skips rebels, pirates, mercenary hosts and conquistador hosts
# (cw_is_real_country - steppe hordes are army-type countries too but stay IN, research 14 F1) before any army loop
# runs (research 04 section 5).
cw_monthly_pulse = {
	trigger = { cw_is_real_country = yes }
	effect = { cw_monthly_pulse_effect = yes }
}

# Rebel countries (is_rebel_country = yes: generic_actions/war_of_the_roses_actions.txt:50,57). The ONLY work the
# mod does for them: refresh cw_rebel_disorder on every rebel army once a month (DESIGN.md 2.2, research 15 1.3) -
# no limits, no HQ placement, no organization pass. Pirates and mercenary-company countries fail the trigger
# and stay untouched.
cw_monthly_rebel_pulse = {
	trigger = { is_rebel_country = yes }
	effect = { cw_rebel_disorder_country = yes }
}

# A revolt / civil war breaks out: the new rebel country gets cw_rebel_disorder at once instead of at the next
# monthly pulse (rebel hosts usually fight within days of rising). The effect guards is_rebel_country itself, so
# a civil war declared by a proper country changes nothing.
# CW TODO verify: do the rebel armies already exist when on_revolt_start / on_civil_war_start fire? If not this
# is a harmless no-op and the monthly pulse covers them within a month.
cw_on_revolt_start = {
	effect = {
		scope:target ?= { cw_rebel_disorder_country = yes }
	}
}

# One-off at game start: every real country gets its first HQ (vanilla uses every_country here too,
# _hardcoded.txt:54-56). Never put every_country into a periodic pulse.
cw_on_game_start = {
	effect = {
		every_country = {
			limit = { cw_is_real_country = yes }
			cw_game_start_country = yes
		}
	}
}

# New age: refresh limits, AI takes new slots, player is told (cw_events.1).
cw_on_new_age = {
	trigger = { cw_is_real_country = yes }
	effect = { cw_new_age_country = yes }
}

# War starts: both named parties and the root (joiner) get an immediate HQ pass and the 15-day ticker.
# Idempotent - see cw_war_start_country. Whether on_war_declared fires once or once per participant is
# unverified, hence scope:actor / scope:recipient are addressed explicitly; participants pulled in without
# a hook are caught by the monthly pulse, which also (re)starts the ticker while at war.
cw_on_war_start = {
	effect = {
		scope:actor ?= { cw_war_start_country = yes }
		scope:recipient ?= { cw_war_start_country = yes }
		root = { cw_war_start_country = yes }
	}
}

# War ends (per side): drop the peace cooldown so the next monthly pulse does a full pass.
cw_on_war_end = {
	trigger = { cw_is_real_country = yes }
	effect = { cw_war_end_country = yes }
}

# After a battle: re-check only the two units involved - no loops (research 04 section 5.3).
# Navies, non-real owners and exempt stacks are filtered inside cw_battle_recheck_unit; "?=" skips a
# losing unit that no longer exists (stack-wiped).
cw_on_battle = {
	effect = {
		scope:actor ?= { cw_battle_recheck_unit = yes }
		scope:target ?= { cw_battle_recheck_unit = yes }
	}
}

# v0.5: an overrun army is taken prisoner - its HQ never changes sides (cw_purge_captured_hqs).
cw_on_overrun = {
	effect = {
		if = {
			limit = { cw_unit_holds_captured_hq = yes }
			cw_purge_captured_hqs = yes
		}
	}
}

# A commander took over a unit (often right after a merge / split by the player): re-check that unit.
cw_on_command = {
	effect = {
		scope:unit ?= { cw_recheck_unit = yes }
	}
}
