###################################
# Concentrated Warfare - headquarters regiments (module hq_core)
#
# Design rule (DESIGN.md 2.1): an HQ gives NO bonus; only the ABSENCE of an HQ is penalised
# (unit modifiers cw_no_hq / cw_no_hq_mild / cw_no_hq_harsh, applied by cw_effects.txt); an army led only by a
# colonial HQ on non-overseas ground keeps its combat strength but gets the siege half (cw_no_hq_siege_only).
#
# Every stat line on a unit type is ADDITIVE on top of its category (unit_types/readme.txt,
# levies/readme.txt:46-50). army_auxiliary has max_strength = 0 (05_army_auxiliary.txt), so
# "max_strength = 0.1" here is exactly 100 men (NUnit.REGIMENT_SIZE = 1000).
# No copy_from on purpose: copying an auxiliary template would add its 50-300 men on top.
#
# Why category = army_auxiliary (research 01 section 8): reserves placement in both battle
# formations, exclude_from_combined_arms = yes, donkey-cart model, auxiliary 2D art, icons,
# prices and GUI breakdown rows all come for free. A new category would need ~10 companion files.
###################################

# Army headquarters: the officer corps of a field army. Available to every country from day one.
cw_hq = {
	category = army_auxiliary			# reserves, never takes frontage; exclude_from_combined_arms comes from the category
	age = "age_1_traditions"			# age icon / template selection only; no upgrades_to => the same regiment all game
	is_special = yes					# gold "special regiment" frame in the army builder (vanilla: 1_uniques_*.txt)
	buildable = yes						# always written explicitly (buildable is not reliably inherited, research 01 section 6)
	levy = no							# a buildable unit must never be a levy (levies/readme.txt:75-77)

	max_strength = 0.1					# 100 men
	combat_power = 0.25					# same as vanilla auxiliaries (00_age_templates_land.txt) => ~0.025 total power, negligible

	construction_demand = cw_hq_construction_demand		# near-zero goods (in_game/common/goods_demand/cw_demands.txt)
	maintenance_demand = cw_hq_maintenance_demand

	country_potential = {
		always = yes
	}
	# no location_potential / location_trigger => recruitable in every owned location
	# no mercenaries_per_location         => no mercenary HQs
	# no upgrades_to                      => never obsolete

	limit = cw_hq_limit					# 1 + imperial slot (>= 300 locations) + min(age slots, owned locations / 75), cap 5; main_menu/common/script_values/cw_limit_values.txt

	gfx_tags = { medium_tag }			# vanilla auxiliaries use medium_tag (2_unlocked_through_tech.txt)
}

# Colonial (expeditionary) headquarters: separate pool per overseas theatre (DESIGN.md 2.4).
# Organizes an army ONLY while it stands on an overseas continent (cw_unit_is_organized, cw_triggers.txt), where
# it also carries the small cw_expeditionary bonus; on non-overseas ground the army fights normally but cannot
# besiege (cw_no_hq_siege_only), so colonial HQs cannot be shipped home as extra siege armies (research 15 2.4).
# Defined in full: inheritance of limit / potentials through copy_from is unverified (research 01 section 13.9).
cw_hq_colonial = {
	category = army_auxiliary
	age = "age_3_discovery"				# icon only; availability is gated below
	is_special = yes
	buildable = yes
	levy = no

	max_strength = 0.1					# 100 men
	combat_power = 0.25

	construction_demand = cw_hq_construction_demand
	maintenance_demand = cw_hq_maintenance_demand

	country_potential = {
		current_age_or_later = { age = age_3_discovery }	# Age of Discovery onwards (syntax: 1_uniques_for_age_3_discovery.txt:114)
		cw_has_overseas_theatre = yes						# owns overseas land or has an expedition overseas (cw_triggers.txt)
	}
	# CW TODO verify (review finding): location_potential runs with root = the LOCATION (0_knights.txt:202), and
	# is_overseas_for_owner is relative to the location OWNER, not the recruiting country (research 10 section 1.3).
	# An overlord may recruit in subject land (research 10 section 2.3), so inside a colonial subject the unit is
	# offered / withheld from the SUBJECT point of view. Test: as an overlord, open the army builder in a colonial
	# subject location overseas and confirm cw_hq_colonial is offered; if not, drop this block and rely on
	# country_potential (cw_has_overseas_theatre) plus the script placement. No vanilla location_potential links
	# back to the recruiting country, so there is no verified alternative.
	location_potential = {
		is_overseas_for_owner = yes		# players recruit it only in overseas locations; the AI gets it by script (cw_ai_assign_hqs)
	}

	limit = cw_colonial_hq_limit		# per-theatre slots, cap 4; main_menu/common/script_values/cw_limit_values.txt

	gfx_tags = { medium_tag }
}
