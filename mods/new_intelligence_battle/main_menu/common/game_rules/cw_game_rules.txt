# Concentrated Warfare - game rules (module warscore_rules, SPEC section 10).
# Format: game/main_menu/common/game_rules/_game_rules.info and 00_game_rules.txt (research 03 section 4.1).
#
# * Every setting carries flag = general_rule, otherwise the rule is invisible in the game-rules UI
#   (in_game/gui/game_rules.gui only lists settings flagged general_rule / flavour_rule / multiplayer_rule).
# * Every setting also carries a flag named like itself. Vanilla script tests settings with has_game_rule = <setting key>
#   (e.g. has_game_rule = institution_spawn_random_city, a setting WITHOUT a self-named flag), and vanilla also adds
#   self-named flags (country_change_allowed, byz_name_*, timeout_*), so has_game_rule = cw_* resolves either way.
# * Loc keys: rule_<rule key>, setting_<setting key>, setting_<setting key>_desc.

# 1. Strength of the "army without headquarters" penalty.
#    Read by hq_core (cw_update_army_organization): has_game_rule = cw_hq_penalty_mild / cw_hq_penalty_harsh pick the
#    unit modifier cw_no_hq_mild / cw_no_hq_harsh instead of cw_no_hq.
cw_rule_hq_penalty = {
	default = cw_hq_penalty_standard

	cw_hq_penalty_standard = {
		flag = cw_hq_penalty_standard
		flag = general_rule
	}
	cw_hq_penalty_mild = {
		flag = cw_hq_penalty_mild
		flag = general_rule
	}
	cw_hq_penalty_harsh = {
		flag = cw_hq_penalty_harsh
		flag = general_rule
	}
}

# 2. (removed) "AI stack concentration" on/off. The game-rule "defines = { }" block documented in _game_rules.info is NOT
#    accepted by the 1.4.7 parser ("Unexpected token: defines", error.log 2026-11-04), so a rule cannot restore the NAI
#    values of loading_screen/common/defines/cw_ai_defines.txt. The define overrides are therefore always active; to play
#    without them, remove that file.

# 3. War score rework: the capital bonus (on_action cw_on_capital_taken fires only with has_game_rule = cw_warscore_on).
#    The NWar define overrides in loading_screen/common/defines/cw_war_defines.txt and the per-rank
#    local_war_score_efficiency in location_ranks/00_default.txt are static data and cannot be toggled by a rule
#    (see 2.); this rule only switches the scripted capital bonus.
cw_rule_warscore = {
	default = cw_warscore_on

	cw_warscore_on = {
		flag = cw_warscore_on
		flag = general_rule
	}
	cw_warscore_off = {
		flag = cw_warscore_off
		flag = general_rule
	}
}

# 4. Native penalty "isolated military tradition" (in_game/common/auto_modifiers/cw_auto_modifiers.txt, scripted trigger
#    cw_has_isolated_tradition in main_menu/common/scripted_triggers/cw_warscore_triggers.txt - menu stage, because the
#    main_menu HQ limit values read it -, decision in_game/common/decisions/cw_native_decisions.txt).
#    Script reads has_game_rule = cw_native_penalty_on.
cw_rule_native_penalty = {
	default = cw_native_penalty_on

	cw_native_penalty_on = {
		flag = cw_native_penalty_on
		flag = general_rule
	}
	cw_native_penalty_off = {
		flag = cw_native_penalty_off
		flag = general_rule
	}
}

# 5. Event "enemy changed its war stance" (cw_stance.1 in in_game/events/cw_stance_events.txt, module ai_stance; the

	}

		flag = general_rule
	}
