# Concentrated Warfare - headquarters limits (module hq_core)
#
# Country-scope script values. They live in main_menu/common/script_values because vanilla keeps
# every unit limit here (default_values.txt:607-690, e.g. wagenburg_unit_limit), and the unit types
# reference them by name: limit = cw_hq_limit (in_game/common/unit_types/cw_headquarters.txt).
# Every desc = "KEY" is a localization key shown in the limit breakdown
# (UNIT_LIMIT_ENTRY_FORMAT: "$KEY$: $VALUE|+=2$"); SUBUNIT_LIMIT_BASE is vanilla ("Base").
#
# Grammar: _script_values.info (value/add/subtract/multiply/divide/min/max/floor, if/else_if with limit).
# "max = X" lowers the value to X if it is higher  => "value = A  max = B" is min(A, B).
# desc inside blocks: add = { value desc } (default_values.txt:665-668), value = { value desc }
# (default_values.txt:676-679, qizilbash_unit_limit), min = { desc value } (in_game script values, grep).
#
# LAYERING (review finding): this main_menu file must NOT reference a scripted trigger defined in the
# in_game layer (no vanilla main_menu file does; main_menu scripted triggers are visible in_game, not
# the other way round). The "isolated military tradition" test is therefore INLINED below as seven engine
# triggers that mirror cw_has_isolated_tradition (module warscore_rules,
# main_menu/common/scripted_triggers/cw_warscore_triggers.txt - a menu-stage trigger since the final
# consistency pass, kept inlined here all the same so that this file never depends on the load order of
# another database) and the potential_trigger of cw_isolated_tradition_auto
# (in_game/common/auto_modifiers/cw_auto_modifiers.txt) - keep all three in sync. Tuning pass (research 15
# section 6 / 7 P3): the former "government_type = government_type:tribe" line is gone (any government type)
# and subjects of a non-American overlord are excluded (is_subject / overlord ?= lines, copied from the
# auto modifier); the verify pass re-synchronised both copies below.
# Engine triggers and in_game database objects are fine here: vanilla main_menu script already
# uses current_age = age_3_discovery, capital ?= { continent = continent:america }, has_variable and
# institution:renaissance (achievements/standard_achievements.txt:289, 488-490, 1017).
# The native rule is always a TRAILING override, so a parse problem with it can at worst disable the
# native rule, never the age / theatre slots.
# CW TODO verify: at the main menu and after loading a game, grep error.log for "cw_age_hq_slots",
# "cw_colonial_hq_limit", "has_game_rule", "is_subject", "overlord" (unknown trigger / value in the main_menu

# v0.4 (user feedback 2026-11-04): headquarters do NOT grow with ages any more, only:
#   * one bonus headquarters from the Age of Reformation ("the age of religious wars") for a country that has
#     embraced the military institution Pike and Shot (institution/age_3_discovery_institutions.txt:100) - so even
#     a small country can field two staffs then; no further age bonus later;
#   * territory: one headquarters per 150 own locations (lands of subjects do not count);
#   * colonies: the separate colonial pool (cw_colonial_hq_limit below).
# The native-penalty rule needs no branch here any more: an American country without Pike and Shot gets no age bonus anyway.
cw_age_hq_slots = {
	value = 0
	if = {
		limit = {
			OR = {
				current_age = age_4_reformation
				current_age = age_5_absolutism
				current_age = age_6_revolutions
			}
			has_embraced_institution = institution:pike_and_shot
		}
		value = 1
	}
}

# one headquarters per 150 own locations (v0.4; v0.3 had 75 and capped it by the age slots)
cw_size_hq_slots = {
	value = num_locations
	divide = 150
	floor = yes
}

# home headquarters limit = 1 + age bonus + territory, cap 5
cw_hq_limit = {
	add = {
		value = 1
		desc = "SUBUNIT_LIMIT_BASE"
	}
	add = {
		value = cw_age_hq_slots
		desc = "CW_HQ_LIMIT_AGE"
	}
	add = {
		value = cw_size_hq_slots
		desc = "CW_HQ_LIMIT_SIZE"
	}
	max = {
		value = 5
		desc = "CW_HQ_LIMIT_CAP"
	}
	min = {
		value = 1
		desc = "CW_HQ_LIMIT_MIN"
	}
}

# Triggers: any_owned_location = { count >= N ... } (flavor_decisions_rus.txt:512-515),
