# Concentrated Warfare - headquarters limits (module hq_core)
#
# Country-scope script values. They live in main_menu/common/script_values because vanilla keeps
# every unit limit here (default_values.txt:607-690, e.g. wagenburg_unit_limit), and the unit types
# reference them by name: limit = cw_hq_limit (in_game/common/unit_types/cw_headquarters.txt).
# Every desc = "KEY" is a localization key shown in the limit breakdown
# (UNIT_LIMIT_ENTRY_FORMAT: "$KEY$: $VALUE|+=2$"); SUBUNIT_LIMIT_BASE is vanilla ("Base").
#
# Grammar: _script_values.info (value/add/subtract/multiply/divide/min/max/floor, if/else_if with limit).
# "max = X" lowers the value to X if it is higher  => "value = A  max = B" is min(A, B).
# desc inside blocks: add = { value desc } (default_values.txt:665-668), value = { value desc }
# (default_values.txt:676-679, qizilbash_unit_limit), min = { desc value } (in_game script values, grep).
#
# LAYERING (review finding): this main_menu file must NOT reference a scripted trigger defined in the
# in_game layer (no vanilla main_menu file does; main_menu scripted triggers are visible in_game, not
# the other way round). The "isolated military tradition" test is therefore INLINED below as seven engine
# triggers that mirror cw_has_isolated_tradition (module warscore_rules,
# main_menu/common/scripted_triggers/cw_warscore_triggers.txt - a menu-stage trigger since the final
# consistency pass, kept inlined here all the same so that this file never depends on the load order of
# another database) and the potential_trigger of cw_isolated_tradition_auto
# (in_game/common/auto_modifiers/cw_auto_modifiers.txt) - keep all three in sync. Tuning pass (research 15
# section 6 / 7 P3): the former "government_type = government_type:tribe" line is gone (any government type)
# and subjects of a non-American overlord are excluded (is_subject / overlord ?= lines, copied from the
# auto modifier); the verify pass re-synchronised both copies below.
# Engine triggers and in_game database objects are fine here: vanilla main_menu script already
# uses current_age = age_3_discovery, capital ?= { continent = continent:america }, has_variable and
# institution:renaissance (achievements/standard_achievements.txt:289, 488-490, 1017).
# The native rule is always a TRAILING override, so a parse problem with it can at worst disable the
# native rule, never the age / theatre slots.
# CW TODO verify: at the main menu and after loading a game, grep error.log for "cw_age_hq_slots",
# "cw_colonial_hq_limit", "has_game_rule", "is_subject", "overlord" (unknown trigger / value in the main_menu

# Extra home HQ slots granted by the age (DESIGN.md 2.3, balance knob):
#   Traditions 0, Renaissance 0, Discovery 1, Reformation 2, Absolutism 3, Revolutions 4.
# Branch pattern: holy_war_values.txt:1-22 (current_age = age_x inside if/else_if).
# Zero while the country lives under the "isolated military tradition" (native penalty, SPEC section 9):
# trailing, order-independent override (see the header).
cw_age_hq_slots = {
	value = 0
	if = {
		limit = { current_age = age_3_discovery }
		value = 1
	}
	else_if = {
		limit = { current_age = age_4_reformation }
		value = 2
	}
	else_if = {
		limit = { current_age = age_5_absolutism }
		value = 3
	}
	else_if = {
		limit = { current_age = age_6_revolutions }
		value = 4
	}
	# isolated military tradition => no extra slots (mirrors cw_has_isolated_tradition, keep in sync)
	if = {
		limit = {
			has_game_rule = cw_native_penalty_on								# main_menu/common/game_rules/cw_game_rules.txt (setting key)
			is_real_country = yes
			capital ?= { continent = continent:america }						# achievements/standard_achievements.txt:289
			OR = {																# not the subject of a non-American overlord: colonial nations / conquistador hosts fight with the overlord's tradition (mirrors cw_is_american_native)
				is_subject = no													# advances/country_son.txt:37
				overlord ?= { capital ?= { continent = continent:america } }	# customizable_localization/country_ranks.txt:26 form; a tributary of an American overlord still counts
			}
			NOT = { has_embraced_institution = institution:pike_and_shot }	# institution/age_3_discovery_institutions.txt:100
			NOT = { has_variable = cw_reformed_host }							# decision cw_reform_war_host (module warscore_rules)
		}
		value = 0
	}
}

# Extra home HQ slots granted by size: one per 75 owned locations (balance knob: the divisor; research 14 R2 /
# 15 section 2.2, maintainer decision: with 100 only 13 regular countries of 1337 owned a single size slot, France
# stayed at 2 HQs for the whole game and a 5th HQ needed 400 locations; with 75 France reaches 3, Castile 4,
# the Yuan / Delhi / Japan / the Horde 5, while 2323 of 2351 countries still live with one army).
# num_locations = the country's own locations (value + trigger, advances/0_age_of_discovery.txt:588); subject
# land does NOT count here (every subject gets its own HQ, research 14 R4).
cw_size_hq_slots = {
	value = num_locations
	divide = 75
	floor = yes
}

# Home HQ limit = 1 (always) + imperial slot + min(age slots, size slots), capped at 5, never below 1 (DESIGN.md 2.3).
#   imperial slot: +1 at >= 300 own locations in EVERY age (the Yuan, the Horde, Delhi, Japan at the 1337 start),
#   so continental empires can police revolts on two fronts before the Age of Discovery (research 15 section 2.2);
#   num_locations >= N as a trigger: decisions/flavor_decisions_lat.txt:105. It is deliberately NOT zeroed by the
#   isolated military tradition (cw_age_hq_slots): no American polity can own 300 locations before the reform.
#   cap 5: a script-value max = { value desc } block, diplomatic_values.txt:310-313 (the colonial cap uses the same).
# => small countries live with one army all game; a great power reaches 5 in the Age of Revolutions.
cw_hq_limit = {
	add = {
		value = 1
		desc = "SUBUNIT_LIMIT_BASE"
	}
	if = {
		limit = { num_locations >= 300 }
		add = {
			value = 1
			desc = "CW_HQ_LIMIT_IMPERIAL"
		}
	}
	add = {
		value = cw_age_hq_slots
		max = cw_size_hq_slots			# min(age, size)
		desc = "CW_HQ_LIMIT_AGE_AND_SIZE"
	}
	max = {
		value = 5
		desc = "CW_HQ_LIMIT_CAP"
	}
	min = {
		value = 1
		desc = "CW_HQ_LIMIT_MIN"
	}
}
