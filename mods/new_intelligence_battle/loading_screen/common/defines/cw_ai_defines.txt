NAI = {
	# --- stack size, splitting, merging (research 06 §2.1) - the concentration group, unchanged by the balance review ---
	RATIO_OVER_SUPPLY_TARGET_TO_SPLIT = 1.2				# 1.3  (l.1026): v0.3 - split a stack sooner when it outgrows the supply limit (user: the AI must minimise attrition; v0.2 had 1.5) (no "split at 130%" churn).
	AI_LAND_UNIT_ATTRITION_UTILITY = -3					# -2   (l.1466): v0.3 - the AI values modifiers that reduce land attrition more (policies, privileges)
														#                1.8 in the first draft: stacks at 180% of the supply target bleed attrition in
														#                mountains / deserts (supply_limit 15, max_attrition 10 - review 15 §5), hence 1.5
	AI_LAND_ARMY_MERGE_THRESHOLD = 0.95					# 0.80 (l.1137): idle armies below 95% frontage fill merge with co-located armies
	HUNT_ARMIES_WIN_CHANCE_SPLIT = 1.0					# 0.99 (l.997) : never split a hunting army just because the win chance is high
	HUNT_ARMIES_MAX_SPLINTERS = 1						# 6    (l.998) : v0.4 - never splinter a hunting army (fragments have no headquarters and fight at half strength)
	HUNT_ARMIES_AI_MAX_NUM_OBJECTIVES = 2				# 5    (l.1004): v0.5 - at most 2 hunting groups at once (maximum concentration, fewer pathfinds)
	CARPET_SIEGE_MAX_ARMIES = 2							# 10   (l.1021): v0.4 - at most 2 siege groups: only armies with a headquarters can siege at all
	AI_CARPET_SIEGE_MAX_ARMIES_FACTOR = 0.05			# 0.1  (l.1593): fewer siege detachments per location to siege
	MAX_RECRUIT_DISTANCE = 6							# 10   (l.1037): reinforcements appear next to the main army and merge sooner
	AI_CARPET_SIEGE_FAILED_PATHFIND_RETRY_DAYS = 14		# 7    (l.835) : v0.5 - a siege group whose path failed retries after 14 days, not 7 (performance)

	# --- objective assignment and churn (§2.2) ---
	AI_REASSIGN_UNIT_THRESHOLD = 1.6					# 1.33 (l.1403): a new objective needs 60% more score to steal a unit - less marching back and forth

	# --- sieges and the covering force (§2.3-2.4) ---
	CARPET_SIEGE_CONFRONT_FACTOR = 0.3					# 0.5  (l.1017): siege detachments avoid battle, the main army fights
	AI_SUPPORT_ARMIES_DESIRED_STRENGTH_FACTOR = 0.33	# 0.33 (l.1592): = VANILLA on purpose. The first draft used 0.5 (research 06 §2.4: half the army
														#                as covering force); the balance review (15 §5) returned it: with only HQ armies able
														#                to siege, idling half the army feeds the white-peace risk (YEARS_SINCE_LAST_WAR_ACTION
														#                _BEFORE_WHITE_PEACE = 3). Concentration comes from the split/merge keys above, not
														#                from idling. Kept as an explicit override so the key stays in the OFF-rule list and
														#                the T4.4 re-tune is a one-number edit.
	AI_ENEMY_STRENGTH_NO_SUPPORT_ARMIES_FACTOR = 2		# 2    (l.1591): = VANILLA on purpose, same reason: a covering force kept at 2-3x superiority only
														#                slowed sieges in wars already won (the first draft used 3). Explicit for the OFF list.

	# --- cooperation with allies and subjects (§2.5) ---
	AI_AWAIT_REINFORCING_ARMY_DISTANCE_MAX = 45			# 35   (l.1131): wait for collaborating armies from further away... (v0.5: 60 -> 45, less waiting and re-planning)
	AI_AWAIT_REINFORCING_ARMY_DISTANCE_MIN = 25			# 20   (l.1132): ...and resume moving once they are this close
	AI_WAIT_FOR_JOINING_UNIT_STRENGTH_THRESHOLD = 0.2	# 0.33 (l.1135): wait even for smaller joining units
	AI_ATTACH_MIN_FRONTAGE_ROOM = 0.5					# 1.0  (l.1139): attach even as reserve depth
	AI_REINFORCE_ONGOING_BATTLE_DISTANCE = 3			# 2    (l.1133) v0.5: 4 -> 3 (performance): nearby armies pile into the decisive battle...
	AI_REINFORCE_BATTLE_DISTANCE_LIMIT = 4				# 3    (l.1587): ...from up to 4 locations away (v0.5: 5 -> 4, performance)
	AI_REINFORCE_BATTLE_STRENGTH_TRESHOLD = 0.7			# 0.55 (l.1589): keep reinforcing until the own side holds 70% of the battle (decisive superiority)
	AI_CALL_ALLY_STRENGTH_BALANCE_THRESHOLD = 0.85		# 0.75 (l.1435): call allies into wars more often

	# --- asymmetric warfare when outnumbered (§2.7; the engine-side "harassing" mode) ---
	HEAVILY_OUTNUMBERED_RATIO = 0.45					# 0.40 (l.1109): outnumbered behaviour kicks in a little earlier. Hedged down from the first draft's
														#                0.5: the StrengthBalance semantics are UNVERIFIED - if it is a fraction like
														#                relative_military_strength, 0.5 = parity and every slightly weaker AI would stop
														#                contesting the field (review 15 §5 / §8, test T5.1). The mod's own harassing stance
														#                (cw_stance_values.txt, 0.33) carries the flavour.
	OUTNUMBERED_BATTLE_ADVANTAGE = 0.5					# 0.60 (l.1110): when outnumbered only pick off stacks of at most 50% of own strength
	AI_ENEMY_LAND_SUPREMACY_THRESHOLD = 2				# 3    (l.1402): in defensive wars hang back and defend already at 2x enemy land strength

	# --- tempo (§2.6) ---
	AI_RECOVER_MORALE_THRESHOLD = 70					# 66   (l.1121): rest below 70% morale - fewer half-broken attacks ("waiting"). 75 in the first draft

	# --- overseas (§2.9) ---
	AI_TRANSPORT_STRENGTH_THRESHOLD_PER_DISTANCE = 0.0003	# 0.0004 (l.828): colonial powers ship troops with somewhat smaller transport fleets
